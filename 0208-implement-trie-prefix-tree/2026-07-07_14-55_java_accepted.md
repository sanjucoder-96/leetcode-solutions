class Node{
   boolean present;
    Node[] c ;

    public Node() {
        this.present = false;
        this.c = new Node[26];
    }
}

class Trie {

       Node root ;

    public Trie() {
        root = new Node();
    }

    public void insert(String word) {
        Node current=root;
        for(char c : word.toCharArray()){
            int val=c-'a';
            if(current.c[val]==null)
                current.c[val]= new Node();
            current=current.c[val];
        }
            current.present=true;
    }

    public boolean search(String word) {
        Node current=root;
        for(char c: word.toCharArray()){
            int val=c-'a';
            if(current.c[val]==null) return false;
            current=current.c[val];

        }
        return current.present;
    }

    public boolean startsWith(String prefix) {
        Node current=root;
        for(char c : prefix.toCharArray()){
            int val= c-'a';
            if(current.c[val]==null) return false;
            current=current.c[val];
        }
        return  true;
    }
}
