## Solution

```java
class Solution {
    public int[][] kClosest(int[][] points, int k) {
          int[][] res = new int[k][2];
        PriorityQueue<int[]> pq = new PriorityQueue<>((a,b)->(b[2]-a[2]));
        int n= points.length;
        for (int i = 0; i <n; i++) {
            int x=points[i][0];
            int y=points[i][1];
            int sq= x*x+y*y;
            pq.add(new int[]{x,y,sq});
            if(pq.size()>k) pq.poll();
        }
        int l=0;
        while(!pq.isEmpty()){
            int[] a=pq.poll();
            res[l][0]=a[0];
            res[l++][1]=a[1];
        }
        return res;
    }
}
```
