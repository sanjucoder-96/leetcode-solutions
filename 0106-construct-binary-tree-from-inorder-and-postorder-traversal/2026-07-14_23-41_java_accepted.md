## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
     public TreeNode solve(int inSt,int inEnd,int postSt,int postEnd,Map<Integer,Integer> mp,int[] inorder,int[] postorder){
        if(inSt>inEnd || postSt>postEnd) return null;
        int ele=postorder[postEnd];
        TreeNode root = new TreeNode(ele);
        int idx=mp.get(ele);
        int leftSubTreeSize = idx-inSt;
        int rightSubTreeSize= inEnd-idx;
        root.left=solve(inSt,idx-1,postSt,postSt+leftSubTreeSize-1,mp,inorder,postorder);
        root.right=solve(idx+1,inEnd,postSt+leftSubTreeSize,postEnd-1,mp,inorder,postorder);
        return root;

    }
    public TreeNode buildTree(int[] inorder, int[] postorder) {
            Map<Integer,Integer> mp = new HashMap<>();
            int n=postorder.length;
        for (int i = 0; i < n; i++) {
            mp.put(inorder[i],i);
        }

        return solve(0,n-1,0,n-1,mp,inorder,postorder);
    }
}
```
