## Solution

```java
class Solution {
    public int minDeletions(int i,int j,StringBuilder s,StringBuilder t,int[][] dp){
        if(i<0) return j+1;
        if(j<0) return i+1;
        if(dp[i][j]!=-1) return dp[i][j];
        if(s.charAt(i)==t.charAt(j)) {
            return minDeletions(i-1,j-1,s,t,dp);
        }
        int delete=1+minDeletions(i-1,j,s,t,dp);
        int delete2=1+minDeletions(i,j-1,s,t,dp);
        return dp[i][j]=Math.min(delete,delete2);
    }
    public int minDistance(String word1, String word2) {
        int m=word1.length(),n=word2.length();

        int[][] dp =new int[m][n];
        for(int[] r: dp) Arrays.fill(r,-1);
        return minDeletions(m-1,n-1,new StringBuilder(word1),new StringBuilder(word2),dp);

    }
}
```
