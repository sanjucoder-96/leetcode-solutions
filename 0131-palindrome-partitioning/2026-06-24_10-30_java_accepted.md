## Solution

```java
class Solution {
    public boolean isPalindrome(String s){
        int i=0,j=s.length()-1;
        while(i<=j){
            if(s.charAt(i)!=s.charAt(j)) return false;
            i++;
            j--;
        }
        return true;
    }
    public void generate(int start,int n,List<String> ds,List<List<String>> res,String s){
        if(start>=n){
            res.add(new ArrayList<>(ds));
            return;
        }

        for(int idx=start;idx<n;idx++){
            String sub=s.substring(start,idx+1);
            if(isPalindrome(sub)){
                ds.add(sub);
                generate(idx+1,n,ds,res,s);
                ds.removeLast();
            }
        }
    }
    public List<List<String>> partition(String s) {
        List<List<String>> res = new ArrayList<>();
        int n=s.length();
        generate(0,n,new ArrayList<>(),res,s);
        return res;
    }
}
```
