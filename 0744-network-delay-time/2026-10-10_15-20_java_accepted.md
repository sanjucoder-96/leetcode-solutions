## Solution

```java
class Solution {
    public int networkDelayTime(int[][] times, int n, int k) {
        List<int[]>[] adj = new ArrayList[n+1];
        for(int i=1;i<=n;i++) adj[i]= new ArrayList<>();
        for(int[] r : times){
            int u=r[0],v=r[1],w=r[2];
            adj[u].add(new int[]{v,w});

        }
        int[] time = new int[n+1];
        Arrays.fill(time,Integer.MAX_VALUE/2);
        time[k]=0;
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt((int[] a)->(a[1])));
        pq.add(new int[]{k,0});
        while(!pq.isEmpty()){
            int f=pq.peek()[0];
            int w=pq.poll()[1];
            if(w>time[f]) continue;

            for(int[] v : adj[f]){
                int node=v[0];
                int dist=v[1];
                if(time[node]>w+dist){
                    time[node]=w+dist;
                    pq.add(new int[]{node,w+dist});
                }
            }
        }
        int ans=0;
        for(int i=1;i<=n;i++){
            if(time[i]==Integer.MAX_VALUE/2) return -1;
            ans=Math.max(ans,time[i]);
        }
        return ans;
    }
}
```
