## Solution

```java
class Solution {
    public String reverseVowels(String s) {
        StringBuilder sb  = new StringBuilder(s);
        int n=s.length();
        int i=0,j=n-1;
        String vowels="aeiouAEIOU";
        while(i<j){
            char a = sb.charAt(i);
            char b = sb.charAt(j);
            if(vowels.indexOf(a)!=-1 && vowels.indexOf(b)!=-1){
                char temp=a;
                sb.setCharAt(i,b);
                sb.setCharAt(j,temp);
                i++;
                j--;
            }
            else if(vowels.indexOf(a)==-1){
                i++;
            }
            else
            j--;

        }
        return sb.toString();
    }
}
```
