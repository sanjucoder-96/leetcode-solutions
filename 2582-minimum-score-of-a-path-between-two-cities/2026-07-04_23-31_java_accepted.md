## Solution

```java
class Solution {
    int min=Integer.MAX_VALUE;
    public void dfs(int u,boolean[] visited,List<int[]>[] adj){
        visited[u]=true;
        for(int[] neighbour : adj[u]){
            int v=neighbour[0],cost=neighbour[1];
            min=Math.min(min,cost);
            if(!visited[v])
            dfs(v,visited,adj);
        }
    }
    public int minScore(int n, int[][] roads) {
        ArrayList<int[]>[] adj = new ArrayList[n+1];
        min=Integer.MAX_VALUE;
        for(int i=1;i<=n;i++){
            adj[i]= new ArrayList<>();
           
            
        }

        for(int[] a : roads){
            int u=a[0];
            int v=a[1],cost=a[2];

            adj[u].add(new int[]{v,cost});
            adj[v].add(new int[]{u,cost});
        }

        boolean[] visited = new boolean[n+1];
        dfs(1,visited,adj);
        return min;
        
    }
}
```
