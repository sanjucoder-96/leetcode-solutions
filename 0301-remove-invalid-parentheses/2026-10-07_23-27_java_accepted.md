## Solution

```java
class Solution {
    int maxLen;
    Set<String> st;

    public void solve(int i, String curr, String s, int count) {
        if (count < 0)
            return;
        if (i == s.length()) {
            if (count == 0) {
                if (curr.length() > maxLen) {
                    st.clear();
                    maxLen = curr.length();
                }
                if (maxLen == curr.length()) {
                    st.add(curr);
                }
            }
            return;
        }
        if (s.charAt(i) != '(' && s.charAt(i) != ')') {
            solve(i + 1, curr + s.charAt(i), s, count);
            return;
        }
        solve(i + 1, curr + s.charAt(i), s, count + (s.charAt(i) == '(' ? 1 : -1));
        solve(i + 1, curr, s, count);
    }

    public List<String> removeInvalidParentheses(String s) {
        st = new HashSet<>();
        maxLen = 0;
        solve(0, "", s, 0);
        return new ArrayList<>(st);
    }
}
```
