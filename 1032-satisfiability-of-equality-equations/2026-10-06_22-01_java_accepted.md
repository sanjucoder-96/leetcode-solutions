class Solution {
    
 static class DSU {
    int max=1;
    int[] size;
    int[] parent;
    DSU(int n){
        this.size= new int[n+1];
        Arrays.fill(size,1);
        this.parent=new int[n+1];
        for(int i=1;i<=n;i++) parent[i]=i;
    }

    public int find(int x){
        if(parent[x]!=x) return parent[x]=find(parent[x]);
        return x;
    }
    public void union(int x,int y){
        int p_x=find(x),p_y=find(y);
        if(p_x==p_y) return;
        if(size[p_x]<size[p_y]){
            parent[p_x]=p_y;
            size[p_y]+=size[p_x];
        }
        else{
            parent[p_y]=p_x;
            size[p_x]+=size[p_y];
        }
        max=Math.max(Math.max(max,size[p_x]), size[p_y]);

    }
}
     public boolean equationsPossible(String[] equations) {

        DSU dsu= new DSU(26);
        int n=equations.length;
        for(String  s: equations){
            if(s.charAt(1)=='='){
                char a =s.charAt(0),b=s.charAt(3);
                int px=dsu.find(a-'a');
                int py = dsu.find(b-'a');
                
                    dsu.union(a-'a',b-'a');
                
            }
        }
        for(String s : equations){
            if(s.charAt(1)=='!'){
                char a =s.charAt(0),b=s.charAt(3);

                int px=dsu.find(a-'a'),py=dsu.find(b-'a');
                if(px==py) return false;
            }
        }
        return true;
    }
}
