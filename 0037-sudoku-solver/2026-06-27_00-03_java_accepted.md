## Solution

```java
class Solution {
    public boolean valid(char[][] board,int r,int c,char ch){
        for(int i=0;i<9;i++){
            if(board[i][c]==ch) return false;

            if(board[r][i]==ch) return false;
            int box_st_row=(r/3)*3;              // (i/3)->gives box number
            int box_st_col=(c/3)*3;
             // (c/3)->gives box number
            // transforming the 2d array into 1d (i/3) for transforming row index to 1d (c%3) for transforming col into 1d
            int checkRow=box_st_row+(i/3);
            int checkCol=box_st_col+(i%3);
            if(board[checkRow][checkCol]==ch) return false;
        }
        return true;
    }
    public boolean solve(char[][] board){
             for(int i=0;i<9;i++ ){
            for(int j=0;j<9;j++){
                if(board[i][j]=='.'){

                    for(char c='1';c<='9';c++){
                        if(valid(board,i,j,c))
                        {
                            board[i][j]=c;
                       if( solve(board)) return true;
                        board[i][j]='.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }
    public void solveSudoku(char[][] board) {
       solve(board);
       return;
    }
}
```
