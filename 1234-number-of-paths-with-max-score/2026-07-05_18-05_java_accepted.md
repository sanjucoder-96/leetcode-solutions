class Solution {
    Integer[][][] dp;
    int MOD= (int)(1e9+7);
    public int[] maxScore(int i,int j,char[][] arr){
        if(i<0 ||j<0 || arr[i][j]=='X') return new int[]{-1,0};
        if(i==0 && j==0) return new int[]{0,1};
        if(dp[i][j][0]!=null) return new int[]{dp[i][j][0],dp[i][j][1]};
        int[] left=maxScore(i-1,j,arr);
        int[] upLeft=maxScore(i-1,j-1,arr);
        int[] right=maxScore(i,j-1,arr);
        int max=Math.max(left[0],Math.max(right[0],upLeft[0]));
        if(max<0) { 
            dp[i][j][0]=-1;
            dp[i][j][1]=0;
           return new int[]{dp[i][j][0],0};}
        int val=(Character.isDigit(arr[i][j]))?arr[i][j]-'0':0;
        int total=0;
        if(left[0]==max) total=(total+left[1])%MOD;
        if(right[0]==max) total=(total+right[1])%MOD;
        if(upLeft[0]==max) total=(total+upLeft[1])%MOD;
        dp[i][j][0]=(max+val)%MOD;
        dp[i][j][1]=total;
        return new int[]{dp[i][j][0],dp[i][j][1]};
    }
     
    public int[] pathsWithMaxScore(List<String> board) {
        int m=board.size(),n=board.get(0).length();
        char[][] arr = new char[m][n];
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
              arr[i][j] =  board.get(i).charAt(j);
            }
        }
        dp = new Integer[m][n][2];
        int[] res = maxScore(m-1,n-1,arr);
        if(res[0]==-1) res[0]=0;

        
       
        return res;
    }
}
