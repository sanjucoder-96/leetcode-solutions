class Solution {
    Integer[][] dp1;
    Integer[][][] dp2;
    int MOD= (int)(1e9+7);
    public int maxScore(int i,int j,char[][] arr){
        if(i<0 ||j<0 || arr[i][j]=='X') return Integer.MIN_VALUE;
        if(i==0 && j==0) return 0;
        if(dp1[i][j]!=null) return dp1[i][j];
        int left=maxScore(i-1,j,arr);
        int upLeft=maxScore(i-1,j-1,arr);
        int right=maxScore(i,j-1,arr);
        int val=(Character.isDigit(arr[i][j]))?arr[i][j]-'0':0;
        
        int calls = Math.max(left,Math.max(upLeft,right));
        if(calls>=0)
            return dp1[i][j]=(val+calls)%MOD;
        return dp1[i][j]=calls%MOD;
    }
     public int maxPaths(int i,int j,char[][] arr,int sum){
        if(i<0 ||j<0 || arr[i][j]=='X' || sum<0) return 0;
        if(i==0 && j==0 && sum==0) return 1;
        if(dp2[i][j][sum]!=null) return dp2[i][j][sum];
        int val=(Character.isDigit(arr[i][j]))?arr[i][j]-'0':0;
        int left=maxPaths(i-1,j,arr,sum-val);
        int upLeft=maxPaths(i-1,j-1,arr,sum-val);
        int right=maxPaths(i,j-1,arr,sum-val);

        return dp2[i][j][sum]=(left%MOD+(upLeft+right)%MOD)%MOD;
    }
    public int[] pathsWithMaxScore(List<String> board) {
        int m=board.size(),n=board.get(0).length();
        char[][] arr = new char[m][n];
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
              arr[i][j] =  board.get(i).charAt(j);
            }
        }
        dp1 = new Integer[m][n];
        
        int ele=maxScore(m-1,n-1,arr);
        if(ele<0) return new int[]{0,0};
        dp2=new Integer[m][n][ele+1];
        int paths=maxPaths(m-1,n-1,arr,ele);
       
        return new int[]{ele,paths};
    }
}
