class Solution {
    Integer[][][] dp;
    int MOD= (int)(1e9+7);
    public int[] maxScore(int i,int j,char[][] arr){
        if(i<0 ||j<0 || arr[i][j]=='X') return new int[]{-1,0};
        if(i==0 && j==0) return new int[]{0,1};
        if(dp[i][j][0]!=null) return new int[]{dp[i][j][0],dp[i][j][1]};
        int[] left=maxScore(i-1,j,arr);
        int[] upLeft=maxScore(i-1,j-1,arr);
        int[] right=maxScore(i,j-1,arr);
        int max=Math.max(left[0],Math.max(right[0],upLeft[0]));
        if(max<0) { 
            dp[i][j][0]=-1;
            dp[i][j][1]=0;
           return new int[]{dp[i][j][0],0};}
        int val=(Character.isDigit(arr[i][j]))?arr[i][j]-'0':0;
        int total=0;
        if(left[0]==max) total=(total+left[1])%MOD;
        if(right[0]==max) total=(total+right[1])%MOD;
        if(upLeft[0]==max) total=(total+upLeft[1])%MOD;
        dp[i][j][0]=(max+val)%MOD;
        dp[i][j][1]=total;
        return new int[]{dp[i][j][0],dp[i][j][1]};
    }

        }
        dp = new Integer[m+1][n+1][2];

         dp[0][0][0]=0;
        dp[0][0][1]=1;

        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(i==0 && j==0) continue;

                if(arr[i][j]=='X') {
                    dp[i][j][0]=-1;
                    dp[i][j][1]=0;
                    continue;
                }

                int cameFromUp=(i>0)?dp[i-1][j][0]:-1;
                int cameFromLeft=(j>0)?dp[i][j-1][0]:-1;
                int cameFromDiag=(i>0 && j>0)?dp[i-1][j-1][0]:-1;

                int total=0;
                int val = Character.isDigit(arr[i][j]) ? arr[i][j] - '0' : 0;
                int max=Math.max(cameFromDiag,Math.max(cameFromUp,cameFromLeft));
                if(max<0){
                    dp[i][j][0]=-1;
                    dp[i][j][1]=0;
                    continue;
                }

                if(i>0 && dp[i-1][j][0]==max) total = (total+dp[i-1][j][1])%MOD;
                if(j>0 && dp[i][j-1][0]==max) total=(total+dp[i][j-1][1])%MOD;
                if(i>0 && j>0 && dp[i-1][j-1][0]==max) total=(total+dp[i-1][j-1][1])%MOD;

                dp[i][j][0] = (max + val) % MOD;
                dp[i][j][1] = total;

}

int finalPaths = dp[m - 1][n - 1][1];
