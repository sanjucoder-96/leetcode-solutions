## Solution

```java
class Solution {
    public int longest(int i, int j,char[] s){
        if(i>j){
           return 0;
        }
        if(i==j) return 1;
        int count=0;
        if(s[i]==s[j]){
        int inside=longest(i+1,j-1,s);
        if(inside==j-i-1){
            count=2+inside;
        }
        }
        return count;
    }
    public String longestPalindrome(String str) {
        int n=str.length();
        int start=0,max=-1;
        char s[]= str.toCharArray();
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                int len=longest(i,j,s);
                if(len>max){
                    start=i;
                    max=len;
                }
            }
        }
        return str.substring(start,start+max);
    }
}
```
