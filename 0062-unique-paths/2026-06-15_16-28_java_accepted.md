## Solution

```java
class Solution {
    public int stepsTotal(int i,int j,int m,int n,int[][] dp){
        if(i==m-1 && j==n-1) return 1;
        if(i>=m || j>=n) return 0;
        if(dp[i][j]!=-1) return dp[i][j];
        return dp[i][j]=stepsTotal(i+1,j,m,n,dp)+stepsTotal(i,j+1,m,n,dp);
    }
    public int uniquePaths(int m, int n) {
        int[][] dp = new int[m+1][n+1];
        for(int[] a : dp)
        Arrays.fill(a,-1);
        return stepsTotal(0,0,m,n,dp);
    }
}
```
