class Solution {
         int threshold=(int)(1e6);
    Integer[][] dp;
    
    public int solve(int i,int twisted,String s,String t){

        if(i==s.length()) return (twisted == 0) ? 0 : threshold;;
    if (dp[i][twisted] != null) return dp[i][twisted];
        char a=(twisted==1)?'0':s.charAt(i);
        char b=t.charAt(i);
        int ans=threshold;
       if(a=='0' && b=='0'){
           int take=solve(i+1,0,s,t);
               int skip=threshold;
          if (i + 1 < s.length()) {
                skip = 2 + solve(i + 1, 1, s, t);
            }
            ans = Math.min(skip, take);
       }
        else if(a=='1' && b=='1'){
            int take=solve(i+1,0 ,s,t);
            int skip = threshold;
            if(i+1<s.length()){
                skip = 2 + solve(i + 1, 1, s, t);
            }
                        ans = Math.min(skip, take);

        }
        else if(a=='1' && b=='0'){
            if(i+1<s.length()){
                int c=(s.charAt(i+1)=='1')?1:2;
                ans = c + solve(i + 1, 1, s, t);
            }

        }
        
        else if(a=='0' && b=='1'){
            int take=1+solve(i+1,0 ,s,t);
            int skip = threshold;
            if(i+1<s.length()){
                skip = 3 + solve(i + 1, 1, s, t);
            }
                        ans = Math.min(skip, take);

        }
        
        return dp[i][twisted]= ans;
       
    }
    public int minOperations(String s1, String s2) {
        int n=s1.length();
        dp = new Integer[n][2];
        int res= solve(0,0,s1,s2);
        return res>=threshold?-1:res;
    
    }
}
