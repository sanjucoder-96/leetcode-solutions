class FreqSegTree{
    long[] seg ;

    int n;

    public FreqSegTree(int n) {
        this.n = n;
        this.seg = new long[4*n];
    }
    public void update(int i,int l,int r,int idx){
        if(l==r){
            seg[i]=1;
            return;
        }
        int mid = l + (r-l)/2;
        if(idx<=mid) update(2*i+1,l,mid,idx);
       else update(2*i+2,mid+1,r,idx);
        seg[i]=seg[2*i+1]+seg[2*i+2];
    }
    public long rangeQuery(int i,int l,int r,int st,int end){
        if(l>end || r<st) return 0;

        if(l>=st && r<=end) return seg[i];
        int mid = l + (r-l)/2 ;
        long left=rangeQuery(2*i+1,l,mid,st,end);
        long right=rangeQuery(2*i+2,mid+1,r,st,end);
        return left+right;
    }
}
class Solution {
      public long goodTriplets(int[] nums1, int[] nums2) {
        int n=nums1.length;
        Map<Integer,Integer> mp  = new HashMap<>(n);
        for(int i=0;i<n;i++){
            mp.put(nums2[i],i);
        }
        long res = 0;
        FreqSegTree st = new FreqSegTree(n);
        st.update(0,0,n-1,mp.get(nums1[0]));
        for(int i=1;i<n;i++)
        {
            int idx=mp.get(nums1[i]);
            long leftCommon=st.rangeQuery(0,0,n-1,0,idx);
            long rightSideElements=n-1-idx;
            long unCommonLeft =i-leftCommon;
            long rightCommon=rightSideElements- unCommonLeft;
            res+=(rightCommon)*(leftCommon);
            st.update(0,0,n-1,idx);

        }
        return res;

    }
}
