class Solution {
       public void BFS(int u,boolean[] visited,List<List<Integer>> adj){
        Deque<Integer> dq = new ArrayDeque<>();
        dq.add(u);
        visited[u]=true;
        while(!dq.isEmpty()){
            int f=dq.poll();
            for(int  v : adj.get(f)){
                if(!visited[v]){
                    dq.add(v);
                    visited[v]=true;
                }
            }
        }
        
    }
    public int makeConnected(int n, int[][] connections) {
        int e=connections.length;
        if(e<n-1) return -1;
        List<List<Integer>> adj = new ArrayList<>();
        for(int i=0;i<n;i++) adj.add(new ArrayList<>());
        for(int[] r : connections){
            int u=r[0],v=r[1];
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        boolean[] visited = new boolean[n];
        int count=-1;
        for(int i=0;i<n;i++){
            if(!visited[i]){
                BFS(i,visited,adj);
                count++;
            }
        }
        return count;
        

    }
}
