## Solution

```java
class Solution {
    int[] seg;
    public void buildTree(int i,int l,int r,int[] arr){
        if(l==r){
            seg[i]=arr[r];
            return;
        }
        int mid = l + (r-l)/2 ;
        buildTree(2*i+1,l,mid,arr);
        buildTree(2*i+2,mid+1,r,arr);
        seg[i]=Math.max(seg[2*i+1],seg[2*i+2]);
    }
    public boolean rangeQuery(int i,int l, int r,int ele){
        if(seg[i]<ele) return false;

        if(l==r){
            seg[i]=-1;
            return true;
        }
        int mid = l + (r-l)/2 ;
        boolean placed=false;
        if(seg[2*i+1]>=ele)
        placed= rangeQuery(2*i+1,l,mid,ele);
        else
        placed= rangeQuery(2*i+2,mid+1,r,ele);

        seg[i]=Math.max(seg[2*i+1],seg[2*i+2]);
        return placed;
    }
    public int numOfUnplacedFruits(int[] fruits, int[] baskets) {
        int unplaced=0,n=baskets.length;

        seg= new int[4*n];
        buildTree(0,0,n-1,baskets);
        for(int ele : fruits){
            if(!rangeQuery(0,0,n-1,ele)) unplaced++;
        }
        return unplaced;
    }
}
```
