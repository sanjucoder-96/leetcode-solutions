class Solution {
    public List<Integer> sequentialDigits(int low, int high) {
        HashSet<Integer> s = new HashSet<>(List.of(12,23,34,45,56,67,78,89,123,234,345,456,567,678,789,1234,2345,3456,4567,5678,6789,12345,23456,34567,45678,56789,123456,234567,345678,456789,1234567,2345678,3456789,12345678,23456789,123456789));
     List<Integer> res = new ArrayList<>();
        for(int ele : s){
                if(ele>=low && ele<=high) res.add(ele);

        }
        Collections.sort(res);
        return res;
    }
}
