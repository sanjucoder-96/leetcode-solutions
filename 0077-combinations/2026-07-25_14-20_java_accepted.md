## Solution

```java
class Solution {
    public void generate(int i,int n,List<Integer> ds ,int k,List<List<Integer>> res){
        if(ds.size()==k){
            res.add(new ArrayList<>(ds));
            return;
        }
        for(int idx=i;idx<=n;idx++){
            ds.add(idx);
            generate(idx+1,n,ds,k,res);
            ds.removeLast();

        }

    }
    public List<List<Integer>> combine(int n, int k) {
        List<List<Integer>> res = new ArrayList<>();
        generate(1,n,new ArrayList<>(),k,res);
        return res;
    }
}
```
