## Solution

```java
class Solution {
    public int minCostConnectPoints(int[][] points) {
              int n=points.length;
        List<int[]>[] adj = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            adj[i]= new ArrayList<>();
        }
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                int[] a= points[i],b=points[j];
                int dist=   Math.abs(a[0]-b[0])+Math.abs(a[1]-b[1]);
                adj[i].add(new int[]{j,dist});
                adj[j].add(new int[]{i,dist});
            }
        }

        boolean[] visited = new boolean[n];
        PriorityQueue<int[]> pq =  new PriorityQueue<>(Comparator.comparing((int[] a)->a[1]));
        int weight=0;
        pq.add(new int[]{0,0});
        while(!pq.isEmpty()){
            int[] a= pq.poll();
            int u=a[0],w=a[1];
            if(visited[u]) continue;
            visited[u]=true;
            weight+=w;
                for(int[] v : adj[u]){
                    int node=v[0],dist=v[1];
                    if(visited[node]) continue;
                    pq.add(new int[]{node,dist});
                }
        }
        return weight;
    }
}
```
