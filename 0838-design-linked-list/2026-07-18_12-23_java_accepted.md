class ListNode{
    ListNode next;
    int val;

    ListNode(int val){
        this.val=val;
    }
}

class MyLinkedList { 
    int size;   
    ListNode head;
    ListNode tail;
    public MyLinkedList() {
        this.head=null;
        this.size=0;
        this.tail=null;
    }
    
    public int get(int index) {
        if(index>=size) return -1;
        ListNode temp=head;
        for(int i=0;i<index;i++){
                temp=temp.next;
        }
        return temp.val;
    }
    
    public void addAtHead(int val) {
        ListNode n= new ListNode(val);
        n.next=head;
        
        head=n;
        size++;
        if(size==1){
            tail=head;
        }
        
    }
    
    public void addAtTail(int val) {
        ListNode t= new ListNode(val);
        size++;
        if(tail==null){
            tail=t;
            head=t;
            return;
        }
        tail.next=t;
        tail=t;
        

    }
    
    public void addAtIndex(int index, int val) {
        if(index==0){
            addAtHead(val);
            return;

        }
        if(index==size){
            addAtTail(val);
            return;
        }
        ListNode temp= head;
        for(int i=0;i<index-1;i++){
            temp=temp.next;

        }
        ListNode next=temp.next;
        ListNode c= new ListNode(val);
        temp.next=c;
        c.next=next;
        size++;


        
    }
    
    public void deleteAtIndex(int index) {
        if(index>=size) return;
        if(index==0){
            head=head.next;
            
            size--;
            if(size==0) tail=null;
            return;
        }

        ListNode temp=head;
        for(int i=0;i<index-1;i++){
            temp=temp.next;
        }
        if(temp.next.next==null){
            tail=temp;
        }
        temp.next=temp.next.next;
        size--;

    }
}

 * int param_1 = obj.get(index);
 * obj.addAtHead(val);
 * obj.addAtTail(val);
 * obj.addAtIndex(index,val);
 * obj.deleteAtIndex(index);
