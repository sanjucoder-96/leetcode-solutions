## Solution

```java
class Solution {
    public boolean canSplit(int[] arr, int max,int k){
        int c=1;
        int currentSum=0;
        for(int ele : arr){
            if(currentSum+ele<=max) currentSum+=ele;
            else{
                currentSum=ele;
                c++;
            }
        }
        return c<=k;
    }
    public int splitArray(int[] arr, int k) {
        int n=arr.length;
        int low=-1,high=0;
        for(int ele: arr) {
            low=Math.max(ele,low);
            high+=ele;
        }
        int ans=0;
        while(low<=high){
            int mid = (low )+ (high-low)/2 ;
            if(canSplit(arr,mid,k)){
                ans=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return low;
    }
}
```
