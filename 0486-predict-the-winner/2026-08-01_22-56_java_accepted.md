## Solution

```java
class Solution {
    int[][] dp;
    public int solve(int i,int j,int[] arr){
        if(i>j) return 0;
        if(dp[i][j]!=-1) return dp[i][j];
        int takeFirst=arr[i]+Math.min(solve(i+2,j,arr),solve(i+1,j-1,arr));
        int takeSecond=arr[j]+Math.min(solve(i+1,j-1,arr),solve(i,j-2,arr));
        return dp[i][j]=Math.max(takeFirst,takeSecond);
    }
    public boolean predictTheWinner(int[] arr) {
        int n=arr.length;
        int sum=0;
        dp = new int[n][n]; 
        for(int[] r: dp)
        Arrays.fill(r,-1);
        for(int i=0;i<n;i++){
            sum+=arr[i];
        }
        int s=solve(0,n-1,arr);

        int rem=sum-s;
        return s>=rem;

    }
}
```
