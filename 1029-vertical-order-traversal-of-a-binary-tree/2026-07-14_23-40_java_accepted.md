## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    class Pair{
        TreeNode node;
        int depth;
        int lvl;
        Pair(TreeNode node,int lvl,int depth){
            this.node=node;
            this.lvl=lvl;
            this.depth=depth;
        }

        public int getNodeVal() {
            return node.val;
        }

        public int getDepth() {
            return depth;
        }

        public int getLvl() {
            return lvl;
        }
    }
    public List<List<Integer>> verticalTraversal(TreeNode root) {
        List<List<Pair>> res = new ArrayList<>();
        Map<Integer,List<Pair>> mp = new TreeMap<>();
        Queue<Pair> q = new LinkedList<>();
        if(root!=null) q.add(new Pair(root,0,0));
        while(!q.isEmpty()){
            Pair p = q.poll();

            TreeNode curr= p.node;
            int depth=p.depth;
            int vlvl=p.lvl;
            mp.computeIfAbsent(vlvl,k->new ArrayList<>());
            mp.get(vlvl).add(new Pair(curr,vlvl,depth));
            if(curr.left!=null) q.add(new Pair(curr.left,vlvl-1,depth+1));
            if(curr.right!=null) q.add(new Pair(curr.right,vlvl+1,depth+1));
        }

        for(int key : mp.keySet()){

            List<Pair> row= mp.get(key);
            row.sort(Comparator.comparingInt(Pair::getDepth).thenComparing(Pair::getLvl).thenComparing(Pair::getNodeVal));
            res.add(row);
        }
        List<List<Integer>> ans = new ArrayList<>();
        for(List<Pair> r : res){

            List<Integer> f =  new ArrayList<>();
            for(Pair c : r){
                f.add(c.node.val);
            }
            ans.add(f);
        }
        return ans;
    }
}
```
