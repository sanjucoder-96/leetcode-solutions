class Solution {
    public int findPath(int i,int j,int m,int n,int[][] arr,int rem){
        if(i>=m || i<0 || j>=n || j<0 || arr[i][j]==-1){
            return 0;
        }
        if(arr[i][j]==2){
            return rem==1?1:0;
        }
        int temp=arr[i][j];
        arr[i][j]=-1;
        rem--;
        int up=findPath(i-1,j,m,n,arr,rem);
        int left=findPath(i,j-1,m,n,arr,rem);
        int right=findPath(i,j+1,m,n,arr,rem);
        int down=findPath(i+1,j,m,n,arr,rem);
        arr[i][j]=temp;
        return up+left+right+down;

    }
    public int uniquePathsIII(int[][] grid) {
        int m=grid.length,n=grid[0].length;
        int rem=0,str=0,stc=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]!=-1) rem++;

                if(grid[i][j]==1){
                    str=i;
                    stc=j;
                }
            }
        }
        return findPath(str,stc,m,n,grid,rem);
    }
}
