class Solution {
    public int findPath(int i,int j,int m,int n,int[][] arr,int rem){
        if(i>=m || j>=n ||i<0 || j<0  || arr[i][j]==-1) return 0;
        if(arr[i][j]==2) {
            return rem==1?1:0;
            }
            int temp=arr[i][j];
        arr[i][j]=-1;
        rem--;
        int up=findPath(i-1,j,m,n,arr,rem);
        int down=findPath(i+1,j,m,n,arr,rem);
        int right=findPath(i,j+1,m,n,arr,rem);
        int left=findPath(i,j-1,m,n,arr,rem);
        arr[i][j]=temp;
        return up+down+right+left;

    } 
    public int uniquePathsIII(int[][] grid) {
        int m = grid.length;
    int n = grid[0].length;

    int startRow = 0;
    int startCol = 0;
    int rem = 0;

    for(int i=0;i<m;i++){
        for(int j=0;j<n;j++){

            if(grid[i][j] != -1)
                rem++;

            if(grid[i][j] == 1){
                startRow = i;
                startCol = j;
            }
        }
    }
        return findPath(startRow,startCol,m,n,grid,rem);
    }
}
