## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int maxLevelSum(TreeNode root) {
        Deque<TreeNode> dq = new ArrayDeque<>();
        if(root!=null) dq.add(root);
        int maxSum=Integer.MIN_VALUE;int lvl=0;
        int c=1;
        while(!dq.isEmpty()){
            int size=dq.size();
            int s=0;
            while(size-- > 0){
                TreeNode a=dq.poll();
                s+=a.val;
                if(a.left!=null) dq.add(a.left);
                if(a.right!=null) dq.add(a.right);
            
            }
           if(maxSum<s){
            maxSum=s;
            lvl=c;
           }
           c++;
        }
        return lvl;
    }
}
```
