class Solution {
    int[] dp;
    int LOW=(int)(-1e5);
    public int play(int i,int[] piles){
        int n=piles.length;
        if(i>=n) return 0;
        if(dp[i]!=LOW) return dp[i]; 
        int stones=0;
        int res=0;
        int take1=piles[i]-play(i+1,piles);
        int take2=Integer.MIN_VALUE,take3=Integer.MIN_VALUE;
        if(i+1<n)
         take2=piles[i]+piles[i+1]-play(i+2,piles);
        if(i+2<n)
         take3=piles[i]+piles[i+1]+piles[i+2]-play(i+3,piles);

        return dp[i]=Math.max(take1,Math.max(take2,take3));
    }
    public String stoneGameIII(int[] stoneValue) {
        int n=stoneValue.length;
        dp =new int[n];
         Arrays.fill(dp,LOW);
        int ans=play(0,stoneValue);
        String res=(ans>0)?"Alice":"Bob";
        if(ans==0) res="Tie";
        return res;
    }
}
