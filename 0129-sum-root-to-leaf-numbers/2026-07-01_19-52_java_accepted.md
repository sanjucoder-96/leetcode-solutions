## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void dfs(TreeNode root,int currentSum,int ans[]){
        if(root==null) return ;
        int dig=0;
        int val=root.val;
        currentSum= currentSum * 10 + val;
        if(root.left==null && root.right==null){
            ans[0]+=currentSum;
            return;
        }
        dfs(root.left,currentSum,ans);
        dfs(root.right,currentSum,ans);
    }
    public int sumNumbers(TreeNode root) {
        int[] ans = {0};
        dfs(root,0,ans);
        return ans[0];
    }
}
```
