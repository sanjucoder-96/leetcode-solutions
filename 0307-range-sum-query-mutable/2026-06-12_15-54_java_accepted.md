class SegmentTree{
    int[] seg;
    int[] arr;
    int n;
    SegmentTree(int[] arr,int n){
        this.seg= new int[4*n];
        this.arr=Arrays.copyOf(arr,arr.length);
        this.n=n;
    }
    public  void buildTree(int i,int l,int r){
        if(l==r){
            seg[i]= arr[l];
            return;
        }
        int mid = (l+r)>>1;
        buildTree(2*i+1,l,mid);
        buildTree(2*i+2,mid+1,r);
        seg[i]=seg[2*i+1]+seg[2*i+2];
    }
    public void updateVal(int i,int idx,int val,int l,int r){
        if(l==r){
            arr[idx]=val;
            seg[i]=val;
            return;
        }
        int mid =(l+r)>>1;
        if(idx<=mid)
        updateVal(2*i+1,idx,val,l,mid);
        else
            updateVal(2*i+2,idx,val,mid+1,r);
        seg[i]=seg[2*i+1]+seg[2*i+2];
    }
    public int rangeSum(int i,int st,int end,int l,int r){
        if(l>end || r<st) return 0;
        else if(l>=st && r<=end){
            return seg[i];
        }
        int mid=(l+r)>>1;
        return rangeSum(2*i+1,st,end,l,mid) + rangeSum(2*i+2,st,end,mid+1,r);
    }
}
public class NumArray {

    private int[] nums;
    private SegmentTree st ;
    public NumArray(int[] nums) {
        this.nums=nums;
        st= new SegmentTree(nums,nums.length);
        st.buildTree(0,0, nums.length-1);

    }

    public void update(int index, int val) {
            st.updateVal(0,index,val,0,nums.length-1);
    }

    public int sumRange(int left, int right) {

return st.rangeSum(0,left,right,0,nums.length-1);
    }
}
