## Solution

class Solution {
    public int lowerBound(int[] arr,int target){
        int n=arr.length;
        int lb=n;
        int low=0,high=n-1;
        while(low<=high){
            int mid = low+ (high-low)/2 ;
            if(arr[mid]>=target){
                lb=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return lb;
    }
    public int minAbsDifference(int[] arr, int goal) {
        int res=Integer.MAX_VALUE;
        int n=arr.length;
        int n1=n/2 , n2=n-n1;
        int[] leftSums = new int[1<<n1];
        int[] rightSums= new int[1<<n2];
    
        for(int i=0;i<(1<<n1);i++){
            int sum=0;
            for(int mask=0;mask<n1;mask++){
                if((i&(1<<mask))!=0) sum+=arr[mask];
            }
            leftSums[i]=sum;
        }
        for(int i=0;i<(1<<n2);i++){
            int sum=0;
            for(int mask=0;mask<n2;mask++){
                if((i&(1<<mask))!=0) sum+=arr[n1+mask];
            }
            rightSums[i]=sum;
        }

        Arrays.sort(rightSums);
        for(int ele: leftSums){
            int target=goal-ele;
            int lb=lowerBound(rightSums,target);
            if(lb<rightSums.length){
                int sum2=rightSums[lb];
                int total=ele+sum2;
                res=Math.min(res,Math.abs(total-goal));
            }
            if(lb>0){
                int sum2=rightSums[lb-1];
                int total=ele+sum2;
                res=Math.min(res,Math.abs(total-goal));
            }
            
        }
        return res;
}
}
