## Solution

```java
class Solution {
    public int[] sumAndMultiply(String s, int[][] queries) {
       int MOD = (int)1e9+7;
        int n=s.length();
        long[] digSum= new long[n];
        int[] numbersUpto = new int[n];
        long[] numbersFormed = new long[n];
        digSum[0]=s.charAt(0)-'0';
        numbersUpto[0]=s.charAt(0)=='0'?0:1;
        numbersFormed[0]=s.charAt(0)-'0';
        int m=1,ele=0;
        for (int i = 1; i <n; i++) {

            digSum[i]=digSum[i-1]+s.charAt(i)-'0';
            if(s.charAt(i)!='0'){
                numbersFormed[i]=((numbersFormed[i-1]*10) + s.charAt(i)-'0')%MOD;
            }
            else{
                numbersFormed[i]=numbersFormed[i-1];
            }
            numbersUpto[i]=numbersUpto[i-1]+((s.charAt(i)=='0')?0:1);
        }
        long[] tens  = new long[n+1];
        long ten=1;
        for(int i=0;i<=n;i++){
            tens[i]=ten;
            ten= (ten*10)%MOD;

        }
        
        int[] res = new int[queries.length];
        for(int i=0;i<queries.length;i++){
            int l=queries[i][0],r=queries[i][1];
            int k=numbersUpto[r] - ((l>=1)?numbersUpto[l-1]:0);
            long numberBef=(l>=1?numbersFormed[l-1]:0);
            long num = (numbersFormed[r] -  numberBef * tens[k]%MOD + MOD)%MOD;
            long sum = (digSum[r] - ((l>=1)?digSum[l-1]:0))%MOD;
            res[i]=(int)((sum*num)%MOD);
        }
        return res;
    }
}
```
