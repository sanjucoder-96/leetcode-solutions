## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    class Pair{
    TreeNode node;
    int lvl;

    public Pair(TreeNode node, int lvl) {
        this.node = node;
        this.lvl = lvl;
    }
}
    public int widthOfBinaryTree(TreeNode root) {
        Deque<Pair> dq = new ArrayDeque<>();
        if(root!=null) dq.add(new Pair(root,0));
        int max=0;
        while(!dq.isEmpty()){
            int size=dq.size();
            int first=dq.peek().lvl;
            int last=0;
            while(size-- > 0){
                Pair p = dq.poll();
                TreeNode node = p.node;
                int lvl=p.lvl-first;
                last=lvl;

                if(node.left!=null) dq.add(new Pair(node.left,2*lvl+1));
                if(node.right!=null) dq.add(new Pair(node.right,2*lvl+2));
                
                
            }

            max=Math.max(last+1,max);
            
        }
        return max;
    }
}
```
