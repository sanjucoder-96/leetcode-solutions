import java.util.Arrays;

 
 class Node {
    char leftChar;
    char rightChar;
    int prefixLen;

    int suffixLen;
    int maxLen;

    public Node(char leftChar, char rightChar, int prefixLen, int suffixLen, int maxLen) {
        this.leftChar = leftChar;
        this.rightChar = rightChar;
        this.prefixLen = prefixLen;
        this.suffixLen = suffixLen;
        this.maxLen = maxLen;
    }
}


 class Segment_Tree {

        Node[] tree;
        StringBuilder s;

    Segment_Tree(String s){
        this.tree= new Node[s.length()*4];
        this.s=new StringBuilder(s);
    }
    public Node merge(Node left, Node right, int leftLen,int rightLen){
            Node res = new Node(left.leftChar,right.rightChar,left.prefixLen,right.suffixLen,1);
            
            if(leftLen==res.prefixLen && left.rightChar==right.leftChar){
                res.prefixLen+=right.prefixLen;

            }
            if(rightLen==res.suffixLen && right.leftChar==left.rightChar){
                res.suffixLen+=left.suffixLen;
            }
            res.maxLen=Math.max(left.maxLen,right.maxLen);
            if(left.rightChar==right.leftChar){
                res.maxLen=Math.max(res.maxLen,left.suffixLen+right.prefixLen);
            }
            return res;


    }
    public void build(int i,int l,int r){
        if(l==r){
            tree[i]= new Node(s.charAt(l),s.charAt(l),1,1,1);
            return;
        }
        int mid= l + (r-l)/2;
        build(2*i+1,l,mid);
        build(2*i+2,mid+1,r);
        tree[i]=merge(tree[2*i+1],tree[2*i+2],mid-l+1,r-(mid+1)+1);
    }

    public void update(int i,int l,int r,int idx,char x){
        if(l==r){
            s.setCharAt(idx,x);
            tree[i]= new Node(x,x,1,1,1);
            return;

        }
        int mid= l + (r-l)/2 ;
        if(idx<=mid)
            update(2*i+1,l,mid,idx,x);
        else
            update(2*i+2,mid+1,r,idx,x);
        tree[i]=merge(tree[2*i+1],tree[2*i+2],mid-l+1,r-(mid+1)+1);

    }

    

}


class Solution {
        public int[] longestRepeating(String s, String queryCharacters, int[] queryIndices) {
        int k= queryIndices.length;
        int[] res = new int[k];
        int n=s.length();
        Segment_Tree st = new Segment_Tree(s);
        st.build(0,0,n-1);
        for(int i=0;i<k;i++){
            st.update(0,0,n-1,queryIndices[i],queryCharacters.charAt(i) );
            res[i]=st.tree[0].maxLen;
        }
        return res;
        
    }
    }
