class Solution {
    public boolean findSafeWalk(List<List<Integer>> grid, int health) {
          List<List<Integer>> res=new ArrayList<>();
       int n=grid.size();
       int m=grid.get(0).size();
       for(int i=0;i<grid.size();i++){

                res.add(new ArrayList<>(Collections.nCopies(m,Integer.MAX_VALUE)));
       }
       int[] di={0,0,1,-1};
       int[] dj={-1,1,0,0};
        Deque<int[]> dq = new ArrayDeque<>();
        res.get(0).set(0,grid.get(0).get(0));
        dq.add(new int[]{0,0});
        while(!dq.isEmpty()){
            int[] a = dq.pollFirst();
            int i=a[0];
            int j=a[1];

            for(int d=0;d<4;d++){
                int new_i=i+di[d],new_j=j+dj[d];

              if(new_i<0 || new_j<0 || new_i>=n || new_j>=m) continue;

              if(res.get(new_i).get(new_j)>grid.get(new_i).get(new_j)+res.get(i).get(j)){
               res.get(new_i).set(new_j,grid.get(new_i).get(new_j)+res.get(i).get(j));
              
              if(grid.get(new_i).get(new_j)==0)
                  dq.addFirst(new int[]{new_i,new_j});
              else 
                  dq.addLast(new int[]{new_i,new_j});
              }
            }

        }
        int x=res.get(n-1)
                .get(m-1);
        return  health-x>=1;
    }
}
