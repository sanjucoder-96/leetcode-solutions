## Solution

```java
class Solution {
    public void solve(int i,int[] arr,List<Integer> ds, List<List<Integer>> res, int n){
        res.add(new ArrayList<>(ds));
        for(int idx=i;idx<n;idx++){
            if(idx>i && arr[idx]==arr[idx-1] ) continue;
            ds.add(arr[idx]);
            solve(idx+1,arr,ds,res,n);
            ds.removeLast();
        }
    }
    public List<List<Integer>> subsetsWithDup(int[] arr) {
        List<List<Integer>> res = new ArrayList<>();
        int n=arr.length;
        Arrays.sort(arr);
        solve(0,arr,new ArrayList<>(),res,n);
        return res;


    }
}
```
