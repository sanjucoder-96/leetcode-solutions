## Solution

```java
class Solution {
    public int digitSum(int num){
        int sum=0;
        while(num>0){
            sum+=num%10;
            num/=10;
        }
        return sum;
    }
        public int minSwaps(int[] arr) {
           int n=arr.length;

        Integer[] copy =new Integer[n];
       for(int i=0;i<n;i++) copy[i]=arr[i];

       Arrays.sort(copy,(a,b)->{
           int s=digitSum(a),t=digitSum(b);
           if(s!=t) return s-t;
           return a-b;
       });
       
       Map<Integer,Integer> mp = new HashMap<>();
       for(int i=0;i<n;i++){
           mp.put(copy[i],i);
       }
       int swaps=0;
       boolean[] visited = new boolean[n];
       for(int i=0;i<n;i++){
           if(visited[i] || i==mp.get(arr[i])) continue;
           
           int j=i,len=0;
           while(!visited[j]){
               visited[j]=true;
               j=mp.get(arr[j]);
               len++;
           }
           swaps+=len-1;
       }
       return swaps;
        }
}
```
