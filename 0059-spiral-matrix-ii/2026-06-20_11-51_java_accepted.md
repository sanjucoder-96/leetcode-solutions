## Solution

```java
class Solution {
    public int[][] generateMatrix(int n) {
        int[][] arr = new int[n][n];
        int minr=0,minc=0,maxr=n-1,maxc=n-1;
        int c=1,total=n*n;
        while(c<=total){
            for(int j=minc;j<=maxc && c<=total;j++){
                arr[minr][j]=c;
                c++;
            }
            minr++;
            for(int i=minr;i<=maxr && c<=total;i++){
                arr[i][maxc]=c;
                c++;
            }
            maxc--;
            for(int j=maxc;j>=minc && c<=total;j--){
                arr[maxr][j]=c;
                c++;
            }
            maxr--;
            for(int i=maxr;i>=minr && c<=total;i--){
                arr[i][minc]=c;
                c++
                ;
            }
            minc++;

        }
        return arr;
    }
}
```
