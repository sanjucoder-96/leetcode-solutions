## Solution

```java
class Solution {
    static int[][] dir={{-1,-1},{-1,0},{-1,1},{0,-1},{1,0},{1,-1},{1,1},{0,1}};
    public int shortestPathBinaryMatrix(int[][] grid) {
        int n=grid.length;
        if(grid[0][0]==1 || grid[n-1][n-1]==1) return -1;
        if(n==1 ) return 1;
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt((int[] a)->(a[2])));
        pq.add(new int[]{0,0,0});
        grid[0][0]=1;
        int count=0;
        BiPredicate<Integer,Integer> isSafe = (a,b)->(a>=0 && b>=0 && a<n && b<n && grid[a][b]!=1);
        int[][] res  = new int[n][n];
        for(int [] r : res) Arrays.fill(r,Integer.MAX_VALUE/2);
        while(!pq.isEmpty()){
            int[] a= pq.poll();
            int x=a[0],y=a[1],d=a[2];
            
        
            if(d>res[x][y]) continue;
            for(int [] r : dir){
                int x_=x+r[0] , y_=y+r[1],w=1;
                if(!isSafe.test(x_,y_)) continue;
                if(d+w<res[x_][y_]) {
                    res[x_][y_]=d+w;
                    grid[x_][y_]=1;
                    pq.add(new int[]{x_,y_,d+w});
                }
            }
            count++;
        }

        return (res[n-1][n-1]==Integer.MAX_VALUE/2)?-1:res[n-1][n-1]+1;

    }
}
```
