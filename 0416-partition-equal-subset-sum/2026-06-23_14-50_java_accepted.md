## Solution

```java
class Solution {
    Boolean[][] dp;
    public boolean solve(int i,int[] arr,int target,int n){
        if(target==0) return true;
        if(i==n || target<0) return false;
        if(dp[i][target]!=null) return dp[i][target];
        boolean take=solve(i+1,arr,target-arr[i],n);
        boolean notTake=solve(i+1,arr,target,n);
        return dp[i][target]=take||notTake;
    }
    public boolean canPartition(int[] arr) {
        int n=arr.length;
        int sum=0;
        for(int ele: arr) sum+=ele;
        if((sum&1)==1) return false;
        dp = new Boolean[n][sum/2 + 1];
        return solve(0,arr,sum/2,n);

    }
}
```
