class Solution {
    public boolean check(int i,int j,String s,int k,int sn,char[][] board){
        int m=board.length,n=board[0].length;
        if(k==sn) return true;
        if(i<0 || j<0 || i>=m || j>=n) return false;
        if(board[i][j]==s.charAt(k)) {
            // System.out.println(board[i][j]);
            char temp=board[i][j];
            board[i][j]='.';
            boolean up=check(i-1,j,s,k+1,sn,board);
            boolean left=check(i,j-1,s,k+1,sn,board);
            boolean right=check(i,j+1,s,k+1,sn,board);
            boolean down=check(i+1,j,s,k+1,sn,board);
            board[i][j]=temp;
            boolean res= up || left || right || down;
            // System.out.println(res);
            return res;
        }
        return false;

    }
    public boolean exist(char[][] board, String word) {
        int m=board.length,n=board[0].length;
        if(m*n<word.length()) return false;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(board[i][j]==word.charAt(0)){
                    if (check(i,j,word,0,word.length(),board))
                    return true;
                }
            }
        }
        return false;
    }
}
