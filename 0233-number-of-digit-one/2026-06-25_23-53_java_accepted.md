## Solution

```java
class Solution {
    static Integer[][][] dp;
    public static int solve(int idx,int tight,String s,int noOfOnes,int n){
        if(idx==n) return noOfOnes;
        if(dp[idx][tight][noOfOnes]!=null) return dp[idx][tight][noOfOnes];
        int ub=(tight==1)?s.charAt(idx)-'0':9;
        int lb=0;
        int res=0;
        for(int dig=lb;dig<=ub;dig++){

            res+=solve(idx+1,(tight==1 && dig==ub)?1:0,s,noOfOnes + ((dig==1)?1:0),n);

        }
        return dp[idx][tight][noOfOnes]=res;
    }
    public static int countDigitOne(int n) {
        String s=String.valueOf(n);
        dp = new Integer[s.length()][2][11];
        return solve(0,1,s,0,s.length());
    }
}
```
