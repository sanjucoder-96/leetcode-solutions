class Solution {
    public int getLength(int[] arr) {
        int n=arr.length;
        int max=1;
        for(int i=0;i<n;i++){
            Map<Integer,Integer> mp = new HashMap<>();
            int mx=1;
            for(int j=i;j<n;j++){
                mp.put(arr[j],mp.getOrDefault(arr[j],0)+1);
                mx=Math.max(mx,mp.get(arr[j]));
            
            if(mp.size()==1) {
                max=Math.max(j-i+1,max);
                continue;
            }
            if(mx%2!=0) continue;
            boolean hasMx=false,hasHalf=false,valid=true;
            for(int f: mp.values()){
                
                   if(f==mx){
                    hasMx=true;
                   }
                   else if(f==mx/2){
                    hasHalf=true;
                   }
                   else{
                    valid=false;
                    break;
                   }
            }
            if(hasMx && hasHalf && valid) {
                            max=Math.max(max,j-i+1);

            }
            
            }
        }
        
    return max;
    }
}
