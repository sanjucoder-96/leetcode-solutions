## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void dfs(TreeNode root,List<Integer> res,int currentLevel){
        if(root==null) return ;
        if(res.size()==currentLevel) res.add(root.val);
        if(res.get(currentLevel)<root.val){
            res.set(currentLevel,root.val);
        }
        dfs(root.left,res,currentLevel+1);
        dfs(root.right,res,currentLevel+1);
        
        
    }
    public List<Integer> largestValues(TreeNode root) {
        List<Integer> res = new ArrayList<>();
        dfs(root,res,0);
        return res;
    }
}
```
