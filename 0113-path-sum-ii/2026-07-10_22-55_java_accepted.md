## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void dfs(TreeNode root,List<Integer> ds,List<List<Integer>> res,int targetSum){
        if(root==null) return ;
        ds.add(root.val);
        if(root.left==null && root.right==null){
            if(targetSum==root.val) {
                res.add(new ArrayList<>(ds));
            }
            ds.removeLast();
            return;
        }
        dfs(root.left,ds,res,targetSum-root.val);
        dfs(root.right,ds,res,targetSum-root.val);

        ds.removeLast();
       


        
    }
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> res  = new ArrayList<>();
        dfs(root,new ArrayList<>(),res,targetSum);
        return res;

    }
}
```
