## Solution

```java
class Solution {
    Boolean[][] dp ;
    public boolean isPalindrome(int i,int j,char[] s){
        if(i>j) return true;
        if(dp[i][j]!=null) return dp[i][j];
        if(s[i]==s[j])  return dp[i][j]=isPalindrome(i+1,j-1,s) ;
        return dp[i][j]=false;
    }
    public int countSubstrings(String str) {
        int n=str.length();
        char[] s= str.toCharArray();
        int count=0;
         dp = new Boolean[n][n];
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                if(isPalindrome(i,j,s)) count++;
            }
        }
        return count;
    }
}
```
