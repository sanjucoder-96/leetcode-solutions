class Solution {
    public boolean isPalindrome(int i,int j,char[] s){
        if(i>j) return true;
        if(s[i]==s[j])  return isPalindrome(i+1,j-1,s) ;
        return false;
    }
    public int countSubstrings(String str) {
        int n=str.length();
        char[] s= str.toCharArray();
        int count=0;
        boolean[][] dp = new boolean[n][n];
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                if(isPalindrome(i,j,s)) count++;
            }
        }
        return count;
    }
}
