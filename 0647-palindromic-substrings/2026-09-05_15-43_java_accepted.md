## Solution

```java
class Solution {
    public int countpal(char[] s, int i,int j){
        int count=0;
        while(i>=0 && j<s.length && s[i--]==s[j++]){
            count++;
        }
        return count;
    }
    public int countSubstrings(String str) {
        int n=str.length();
        char[] s= str.toCharArray();
        int count=0;
        for(int i=0;i<n;i++){
            count+=countpal(s,i,i);
            if(i+1<n)
            count+=countpal(s,i,i+1);
        }
        return count;
    }
}
```
