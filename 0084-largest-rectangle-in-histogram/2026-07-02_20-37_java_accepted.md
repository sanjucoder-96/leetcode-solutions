## Solution

```java
class Solution {
    public int[] getPreviousSmaller(int[] arr){
        int n=arr.length;
        int[] res = new int[n];
        Deque<Integer> st = new ArrayDeque<>();
        for(int i=0;i<n;i++){
            while(!st.isEmpty() && arr[st.peek()]>=arr[i]) st.pop();
            res[i]=st.isEmpty()?-1:st.peek();
            st.push(i);
        }
        return res;
    }
    public int[] getNextSmaller(int[] arr){
        int n=arr.length;
        int[] res = new int[n];
        Deque<Integer> st = new ArrayDeque<>();
        for(int i=n-1;i>=0;i--){
            while(!st.isEmpty() && arr[st.peek()]>arr[i]) st.pop();
            res[i]=st.isEmpty()?n:st.peek();
            st.push(i);
        }
        return res;
    }
    public int largestRectangleArea(int[] heights) {
        int[] pse = getPreviousSmaller(heights);
        int[] nse = getNextSmaller(heights);

        int maxArea=0;
        int n=heights.length;
        for(int i=0;i<n;i++){
            maxArea=Math.max(maxArea,(nse[i]-pse[i]-1)*heights[i]);
        }
        return maxArea;
    }
}
```
