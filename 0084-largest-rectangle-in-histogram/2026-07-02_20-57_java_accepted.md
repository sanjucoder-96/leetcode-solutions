## Solution

```java
class Solution {
   
    public int largestRectangleArea(int[] arr) {
       int n=arr.length;
        int[] res = new int[n];
        int maxArea=0;
        Deque<Integer> st = new ArrayDeque<>();
        for(int i=0;i<n;i++){
            while(!st.isEmpty() && arr[st.peek()]>=arr[i]){
                
                int x=st.pop();
                int nse=i;
                int pse=(st.isEmpty())?-1:st.peek();
                maxArea=Math.max(maxArea,(nse-pse-1)*arr[x]);

                }
            st.push(i);
        }

        while(!st.isEmpty()){
            int nse = n;
            int ele=arr[st.pop()];
            int pse=(st.isEmpty())?-1:st.peek();
                            maxArea=Math.max(maxArea,(nse-pse-1)*ele);


        }
        return maxArea;
    }
}
```
