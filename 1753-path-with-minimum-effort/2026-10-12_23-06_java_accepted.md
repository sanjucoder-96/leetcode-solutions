## Solution

```java
class Solution {
    static int[][] dir={{0,1},{0,-1},{1,0},{-1,0}};
    public boolean canTrace(int[][] arr,int k){
        int m=arr.length,n=arr[0].length;
        
        Deque<int[]> dq = new ArrayDeque<>();
        dq.add(new int[]{0,0});
        boolean[][] visited = new boolean[m][n];
        visited[0][0]=true;
        while(!dq.isEmpty()){
            int[] f = dq.poll();
            int a=f[0],b=f[1];
            if(a==m-1 && b==n-1) return true;
            for(int[] d : dir){
                int x_=d[0]+a,y_=d[1]+b;
                if(x_<0 || y_<0 || x_>=m || y_>=n || visited[x_][y_]) continue;
                int diff=Math.abs(arr[a][b]-arr[x_][y_]);
                if(diff>k) continue;
                dq.add(new int[]{x_,y_});
                visited[x_][y_]=true;
                
                
            }
        }
        return false;
        
    }
    public int minimumEffortPath(int[][] heights) {
        int m=heights.length,n=heights[0].length;
        if (m == 1 && n == 1)
            return 0;
        int low=0,high=(int)(1e6);
        int ans=(int)(1e6);

        while(low<=high){
            int mid = low+ (high-low)/2;
            if(canTrace(heights,mid)){
                ans=mid;
                high=mid-1;
            }
            else{
                low=mid+1;
            }
        }
        return ans;
    }
}
```
