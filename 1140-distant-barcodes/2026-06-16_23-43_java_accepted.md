## Solution

```java
class Solution {
    public int[] rearrangeBarcodes(int[] arr) {
           int n=arr.length;
        int[] res = new int[n];
        if(n==1) {
            res[0]=arr[0];
            return res;
        }
         Map<Integer,Integer> mp =new HashMap<>();
        for(int ele : arr) mp.put(ele,mp.getOrDefault(ele,0)+1);
        PriorityQueue<Map.Entry<Integer,Integer>> pq = new PriorityQueue<>(Comparator.comparing((a)->-a.getValue()));
        int l=0;
        pq.addAll(mp.entrySet());
       

        while(pq.size()>1){
            Map.Entry<Integer, Integer> a = pq.poll();
            Map.Entry<Integer, Integer> b = pq.poll();
            res[l++]=a.getKey();
            res[l++]=b.getKey();
            
            if(a.getValue()-1>0 ) {
                a.setValue(a.getValue()-1);
                pq.offer(a);
            }
             if(b.getValue()-1>0 ) {
                b.setValue(b.getValue()-1);
                pq.offer(b);
            }
             
    
        }
        while(!pq.isEmpty()) res[l++]=pq.poll().getKey();
        return  res;

    }
}
```
