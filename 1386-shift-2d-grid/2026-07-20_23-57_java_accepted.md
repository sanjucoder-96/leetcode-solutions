## Solution

```java
class Solution {
    public List<List<Integer>> shiftGrid(int[][] arr, int k) {
        int row=arr.length,col=arr[0].length;

        int n=row*col;
        k%=n;
        BiConsumer<Integer,Integer> fun = (i,j)->{

            while(i<j){
            int r1=i/col;
            int c1=i%col;

            int r2=j/col;
            int c2=j%col;

            int temp=arr[r1][c1];
            arr[r1][c1]=arr[r2][c2];
            arr[r2][c2]=temp;

            i++;
            j--;}
        };
        fun.accept(0,n-1);
        fun.accept(0,k-1);
        fun.accept(k,n-1);
          List<List<Integer>> result = new ArrayList<>();
        for (int[] raw : arr) {
            List<Integer> l = new ArrayList<>();
            for (int val : raw) {
                l.add(val);
            }
            result.add(l);
        }
        return result;
    }
}
```
