## Solution

```java
class Solution {
    public boolean hasAllChars(int[] f){
     return (f[0]>=1 && f[1]>=1 && f[2]>=1);
    }
    public int numberOfSubstrings(String s) {
        int n=s.length();
        int l=0;
        int total=0;
        int[] freq = new int[3];
        for(int r=0;r<n;r++){
            
            freq[s.charAt(r)-'a']++;
            while(hasAllChars(freq)){
                total+=n-r;
                freq[s.charAt(l)-'a']--;
                l++;

            }

            
        }
        return total;
    }
}
```
