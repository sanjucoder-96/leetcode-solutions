## Solution

```java
class Solution {
    public int[] searchRange(int[] arr, int target) {
        int n=arr.length;
        if(n==0) return new int[]{-1,-1};
        int low=0,high=n-1;
        int ans=0;
        while(low<=high){
            int mid = low +  (high-low)/2 ;
            if(arr[mid]>=target){
                ans=mid;
                high=mid-1;
            }
            else{
                low=mid+1;
            }
        }
        if(arr[ans]!=target) return new int[]{-1,-1};
        low=0;
        high=n-1;
        int lp=-1;
        while(low<=high)
        {
            int mid = low + (high-low)/2 ;
            if(arr[mid]<=target){
                lp=mid;
                low=mid+1;
            }
            else
            high=mid-1;
        }
        return new int[]{ans,lp};
    }
}
```
