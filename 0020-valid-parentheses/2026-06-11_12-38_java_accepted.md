## Solution

```java
class Solution {
    public boolean check_match(char c,char top){
        return (c==')' && top=='(') || (c=='}' && top=='{') || (c==']' && top=='[');
    }
    public boolean isValid(String s) {
        Deque<Character> st = new ArrayDeque<>();
        for(char c : s.toCharArray()){
            if(c=='(' || c=='{' || c=='[') 
            st.push(c);
            else if(!st.isEmpty() && check_match(c,st.peek()))
                st.pop();
            else
                return false;
        }
        return st.isEmpty();
    }
}
```
