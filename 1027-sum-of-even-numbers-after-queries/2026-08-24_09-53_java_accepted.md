## Solution

```java
class Solution {
    public int[] sumEvenAfterQueries(int[] arr, int[][] queries) {
        int n=arr.length;
        int sum=0;
        boolean[] isEven= new boolean[n];

        for(int i=0;i<n;i++){
            int ele=arr[i];
            if(ele%2==0){
                sum+=ele;
                isEven[i]=true;
            }
        }
        int[] res = new int[queries.length];
        int k=0;
        for(int[] q : queries){
            int idx=q[1],val=q[0];
            if(isEven[idx]){
                if((arr[idx]+val)%2==0){
                    System.out.println(val+" "+(arr[idx]+val));
                    sum+=val;
                }
                else{
                    sum-=arr[idx];
                    isEven[idx]=false;
                }
            }
            else{

                if((arr[idx]+val)%2==0){
                    sum+=arr[idx]+val;
                    isEven[idx]=true;
                }
                
                
            }
            arr[idx]+=val;
            // System.out.println(Arrays.toString(arr));

            res[k++]=sum;
            
        }
        return res;
    }
}
```
