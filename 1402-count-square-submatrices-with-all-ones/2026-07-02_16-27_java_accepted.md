## Solution

```java
class Solution {
    int[][] dp;
    public int solve(int i, int j, int[][] arr, int m, int n) {
        if (i >= m || j >= n || arr[i][j] == 0)
            return 0;
            if(dp[i][j]!=-1) return dp[i][j];

        int right = solve(i, j + 1, arr, m, n);
        int down = solve(i + 1, j, arr, m, n);
        int diag = solve(i + 1, j + 1, arr, m, n);

        return dp[i][j] = arr[i][j] + Math.min(right, Math.min(down, diag));
    }

    public int countSquares(int[][] arr) {
        int m = arr.length;
        int n = arr[0].length;
        int s = 0;
        dp = new int[m][n];
        for(int[] r: dp){
            Arrays.fill(r,-1);
        }
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++)
            s+=solve(i,j,arr,m,n);
        }
        return s;
    }
}
```
