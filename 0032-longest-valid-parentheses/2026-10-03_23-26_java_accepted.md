## Solution

```java
class Solution {
    public int longestValidParentheses(String s) {
        int open=0,close=0;
        int max=0;
        char[] str = s.toCharArray();
        int n=str.length;

        for(int i=0;i<n;i++){
            if(str[i]=='(') open++;
            else close++;
            if(open==close) {
                max=Math.max(open+close,max);
            }
           else if(close>open){
                                open=close=0;

            }
        }
        open=close=0;

        for(int  i=n-1;i>=0;i--){
            if(str[i]=='(') open++;
            else close++;

            if(open==close)
            max=Math.max(max,open+close);
            else if(open>close){
                open=close=0;
            }
        }
        return max;
    }
}
```
