## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode solve(int inSt,int inEnd,int preSt,int preEnd,Map<Integer,Integer> mp,int[] preorder,int[] inorder){
        if(inSt>inEnd || preSt>preEnd) return null;
        
        int ele=preorder[preSt];
        TreeNode root= new TreeNode(ele);
        int idx=mp.get(ele);
        int leftSubTreeSize= idx-inSt;
        int rightSubTreeSize=inEnd-idx;
        root.right=solve(idx+1,inEnd,preSt+leftSubTreeSize+1,preEnd,mp,preorder,inorder);
        root.left=solve(inSt,idx-1,preSt+1,preSt+leftSubTreeSize,mp,preorder,inorder);
        return root;
    }
    public TreeNode buildTree(int[] preorder, int[] inorder) {
         Map<Integer,Integer> mp = new HashMap<>();
         int n=inorder.length;
        for(int i=0;i<n;i++) mp.put(inorder[i],i);
        return solve(0,n-1,0,n-1,mp,preorder,inorder);

    }
}
```
