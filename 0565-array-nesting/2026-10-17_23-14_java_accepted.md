## Solution

```java
class Solution {
    public int minSwaps(int[] arr){
        int res=0;
        int n=arr.length;
        Map<Integer,Integer> mp  = new HashMap<>();
        int[] copy = arr.clone();
        for(int i=0;i<n;i++) mp.put(i,copy[i]);
        boolean[] visited = new boolean[n];
        for(int i=0;i<n;i++){
            if(visited[i]) continue;
            int j=i,len=0;
            while(!visited[j]){
                visited[j]=true;
                j=mp.get(j);
                len++;
            }
            res=Math.max(len,res);
        }
        return res;
    }
    public int arrayNesting(int[] nums) {
        return minSwaps(nums);
    }
}
```
