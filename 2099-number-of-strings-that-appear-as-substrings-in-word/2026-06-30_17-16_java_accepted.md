class Solution {
     public static void buildLPS(char[] str,int[] lps){
        int len=0;
        int i=1,n=str.length;
        lps[0]=0;
        while(i<n){
            if(str[i]==str[len]){
                len++;
                lps[i]=len;
                i++;
            }
            else{
                if(len>0) len=lps[len-1];
                else i++; 
            }
        }
        return;
  }
  public static int match(String pat, String txt){
     int n=pat.length(),m=txt.length();
        // longest proper prefix that matches suffix
        int[] lps = new int[n];
        buildLPS(pat.toCharArray(),lps);
        int i=0,j=0,c=0;
        while(i<m){
            if(pat.charAt(j)==txt.charAt(i)){
                i++;
                j++;
                if(j==n){
                    c++;
                j=lps[j-1];
                }
            }
            else{

                if(j>0) j=lps[j-1];
                else i++;
            }


        }
return c>0?1:0;  }
    public int numOfStrings(String[] patterns, String word) {
       int c=0;
       for(String pat : patterns){
        c+=match(pat,word);
       }
       return c;
    }
}
