## Solution

```java
class Solution {
    public int longestBalanced(String s) {
        int n=s.length();
        Map<Character,Integer> mp ;
        int max=0;
        for(int i=0;i<n;i++){
            mp= new HashMap<>();
            for(int j=i;j<n;j++){
                char c = s.charAt(j);
                mp.put(c,mp.getOrDefault(c,0)+1);
                boolean found=true;
                int freq=  mp.get(c);
                for(char k : mp.keySet()){
                    if(mp.get(k)!=freq){
                        found=false;
                        break;
                    }
                }
                if(found){
                    max=Math.max(max,j-i+1);
                }
            }
        }
        return max;
    }
}
```
