## Solution

```java
class Solution {
    public List<Integer> largestDivisibleSubset(int[] arr) {
        int n=arr.length;
        int[] dp = new int[n];
        Arrays.sort(arr);
        int[] prevIndex=new int[n];
        Arrays.fill(prevIndex,-1);
        int maxL=0;
        int start=0;
        for(int i=0;i<n;i++){
            dp[i]+=1;
            for(int j=0;j<i;j++){
                if(arr[i]%arr[j]==0 && dp[i]<dp[j]+1){
                    dp[i]=dp[j]+1;
                    prevIndex[i]=j;
                if(dp[i]>maxL){
                    maxL=dp[i];
                    start=i;
                }
                }
            }
        }
        
        List<Integer> res = new ArrayList<>();
        int i=start;
        while(i!=-1){
            res.add(arr[i]);
            i=prevIndex[i];
        }
        return res;

    }
}
```
