## Solution

```java
class Solution {
    public long form(int i,int[] arr,int n,long[][] dp,int flag){
        if(i>=n) return 0;
        if(dp[i][flag]!=-1) return dp[i][flag];
        int val=arr[i];
        if(flag==1)
            val=-val;
        long take=val+form(i+1,arr,n,dp,1-flag);
        long skip=form(i+1,arr,n,dp,flag);
        
        return dp[i][flag]= Math.max(take,skip);
                
    }
    public long maxAlternatingSum(int[] arr) {
        int n=arr.length;
        long[][] dp =new long[n+1][2];
        for(long[] r: dp){
            Arrays.fill(r,-1);
        }
        return form(0,arr,n,dp,0);
    }
}
```
