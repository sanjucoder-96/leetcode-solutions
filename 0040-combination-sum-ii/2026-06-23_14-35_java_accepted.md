```java
class Solution {
    public void solve(int idx,int[] arr,int target,List<Integer> ds,List<List<Integer>> res,int n){
        if(target==0) {
                res.add(new ArrayList<>(ds));
                return;
        }
        if(idx>=n) {
                return;
        }

        for(int i=idx;i<n;i++){
            if(i>idx && arr[i]==arr[i-1]) continue;
            if(arr[i] > target)
                break;

            ds.add(arr[i]);
            solve(i+1,arr,target-arr[i],ds,res,n);
            ds.removeLast();
        }


    }
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        List<List<Integer>> res = new ArrayList<>();
        Arrays.sort(candidates);
        int n=candidates.length;
        solve(0,candidates,target,new ArrayList<>(),res,n);
        return res;


    }
}
```
