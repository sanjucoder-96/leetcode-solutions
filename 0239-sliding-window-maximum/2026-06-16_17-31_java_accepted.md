## Solution

```java
class Solution {
    public int[] maxSlidingWindow(int[] arr, int k) {
        int n=arr.length;
        int[] res = new int[n-k+1];
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt((int[] a)->-a[0]));
        for(int i=0;i<k;i++){
            pq.add(new int[]{arr[i],i});
        }
        int l=0;
        res[l++]=pq.peek()[0];
        for(int i=k;i<n;i++){
            pq.add(new int[]{arr[i],i});

            while(!pq.isEmpty() && pq.peek()[1]<=i-k){
                pq.poll();
            }

            res[l++]=pq.peek()[0];
        }


        return res;
    }
}
```
