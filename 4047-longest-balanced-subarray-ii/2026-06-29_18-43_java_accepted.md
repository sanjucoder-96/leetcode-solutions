class Solution {
    int[] segMin;
    int[] segMax;
    int[] lazy;
    public void propagate(int i,int l,int r){
        if(lazy[i]!=0){
            segMin[i]+=lazy[i];
            segMax[i]+=lazy[i];
            if(l!=r){
            lazy[2*i+1]+=lazy[i];
            lazy[2*i+2]+=lazy[i];}
            lazy[i]=0;
        }
    }
    public void updateRange(int i,int l,int r ,int start,int end,int val){
        propagate(i,l,r);
        if(l>end || r<start) return ;
        if(l>=start && r<=end) {
            lazy[i]+=val;
            propagate(i,l,r);
            return;
        }
        int mid= l + (r-l)/2 ;
        updateRange(2*i+1,l,mid,start,end,val);
        updateRange(2*i+2,mid+1,r,start,end,val);
        segMin[i]=Math.min(segMin[2*i+1],segMin[2*i+2]);
        segMax[i]=Math.max(segMax[2*i+2],segMax[2*i+1]);
        
        
    }
    public  int findLeftMostZero(int i,int l,int r){
        propagate(i,l,r);
        if(segMin[i]>0 || segMax[i]<0) return -1;
         if(l==r) return l;
         int mid = l + (r-l)/2;
         int left=findLeftMostZero(2*i+1,l,mid);
         if(left!=-1) return left;
         return findLeftMostZero(2*i+2,mid+1,r);
    }
    public int longestBalanced(int[] arr) {
         int n=arr.length;
        int len=0;
        int[] prefix= new int[n];
        segMin= new int[4*n];
        segMax= new int[4*n];
        lazy= new int[4*n];

        Map<Integer,Integer> mp = new HashMap<>();
        for(int r=0;r<n;r++){
            int val=(arr[r]%2==0)?1:-1;
            int prev=mp.getOrDefault(arr[r],-1);
            if(prev!=-1){
               updateRange(0,0,n-1,0,prev,-val);
            }
            updateRange(0,0,n-1,0,r,val);
            int left=findLeftMostZero(0,0,n-1);
            if(left!=-1)
            len=Math.max(len,r-left+1);

            mp.put(arr[r],r);
        } 
        return len;
    }
}
