## Solution

```java
/*
// Definition for a Node.
class Node {
    public int val;
    public Node left;
    public Node right;
    public Node next;

    public Node() {}
    
    public Node(int _val) {
        val = _val;
    }

    public Node(int _val, Node _left, Node _right, Node _next) {
        val = _val;
        left = _left;
        right = _right;
        next = _next;
    }
};
*/

class Solution {
    public Node connect(Node root) {
        Deque<Node> dq = new ArrayDeque<>();
        if(root!=null)
        dq.add(root);
        int l=0;
        while(!dq.isEmpty()){
            int size=dq.size();
            int max=1<<l,c=1;
            while(size-- > 0){
                Node a  = dq.poll();
                if(c<max)
                a.next=(dq.isEmpty())?null:dq.peek();
                
                if(a.left!=null) dq.add(a.left);
                if(a.right!=null) dq.add(a.right);
                c++;
            }
            l++;
        }
        return root;
    }
}
```
