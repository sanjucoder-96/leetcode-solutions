## Solution

```java
class Solution {
    public int coinChange(int[] coins, int amount) {
        int n=coins.length;
        int[] dp = new int[amount+1];
        Arrays.fill(dp,(int)(1e5));
        dp[0]=0;
        for(int i=coins[0];i<=amount;i++){
            if(i%coins[0]==0){
            dp[i]=(int)(i/coins[0]);}
        }
        for(int i=1;i<n;i++){
            for(int j=coins[i];j<=amount;j++){
               
                    dp[j]=Math.min(1+dp[j-coins[i]],dp[j]);
                
            }
        }
       int ans=dp[amount];
       return ans>=1e5?-1:ans;
    }
}
```
