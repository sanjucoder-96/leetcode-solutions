```java
class Solution {
    int[][] dp ;
    public int getMinCoins(int i,int[] coins,int amount){
        if(amount==0){
            return 0;
        }
        if(i>=coins.length || amount<0 ) return (int)(1e5);
        if(dp[i][amount]!=-1) return dp[i][amount];
        int skip=getMinCoins(i+1,coins,amount);
        if(coins[i]>amount) return dp[i][amount]=skip;
        int take = 1+getMinCoins(i,coins,amount-coins[i]);
        return dp[i][amount]=Math.min(take,skip);
    }
    public int coinChange(int[] coins, int amount) {
        int n=coins.length;
        dp = new int[n][amount+1];
        for(int[] r : dp)
        Arrays.fill(r,-1);
       int ans=getMinCoins(0,coins,amount);
       return ans==(int)(1e5)?-1:ans;
    }
}
```
