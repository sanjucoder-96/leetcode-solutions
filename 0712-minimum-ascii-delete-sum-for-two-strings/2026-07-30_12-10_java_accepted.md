## Solution

```java
class Solution {

    public int minimumDeleteSum(String s, String t) {
        int m=s.length(),n=t.length();
        int[] sPref = new int[m+1];
        int[] tPref= new int[n+1];
        for(int i=1;i<=m;i++){
            sPref[i]=(int)(s.charAt(i-1))+sPref[i-1];
        }
        for(int j=1;j<=n;j++){
            tPref[j]=(int)(t.charAt(j-1))+tPref[j-1];
        }

        int[][] dp = new int[m+1][n+1];

        for(int i=1;i<=m;i++) dp[i][0]=sPref[i];

        for(int j=1;j<=n;j++) dp[0][j]=tPref[j];
        
        for(int i=1;i<=m;i++){
            for(int j=1;j<=n;j++){
                if(s.charAt(i-1)==t.charAt(j-1)){
                    dp[i][j]=dp[i-1][j-1];
                }
                else{
                    int a=(int)(s.charAt(i-1))+dp[i-1][j];
                    int b= (int)(t.charAt(j-1))+dp[i][j-1];
                    
                    dp[i][j]=Math.min(a,b);
                }
            }
        }
        return dp[m][n];
    }
}
```
