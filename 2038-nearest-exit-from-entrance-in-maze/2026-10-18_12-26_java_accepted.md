## Solution

```java
class Solution {
    int m,n;
    public boolean check(int i,int j){
        return i==m-1 || j==n-1 || i==0 || j==0 ;
    }
    
    int[][] dir={{1,0},{-1,0},{0,-1},{0,1}};
    public int nearestExit(char[][] maze, int[] entrance) {
       m= maze.length;n=maze[0].length;
       Deque<int[]> dq = new ArrayDeque<>();
       BiPredicate<Integer,Integer> valid= (a,b)->(a>=0 && b>=0 && a<m && b<n &&  maze[a][b]!='+');
       dq.add(new int[]{entrance[0],entrance[1]});
       maze[entrance[0]][entrance[1]]='+';
       int count=0;
       while(!dq.isEmpty()){
            int size=dq.size();

            while(size-- > 0){
                int[] front= dq.poll();
                int i=front[0],j=front[1];
               
                if((i!=entrance[0] || j!=entrance[1]) && (i==0 || j==0 || i==m-1 || j==n-1 )) return count;
               
                
                for(int[] d :dir){
                   int x=i+d[0],y=j+d[1];
                    if(!valid.test(x,y)) continue;
                    
                    dq.add(new int[]{x,y});
                     maze[x][y]='+';
                }

            }
            count++;
       }
       return -1;
        
    }
}
```
