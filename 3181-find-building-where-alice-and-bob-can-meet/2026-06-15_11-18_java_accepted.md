class Segment_tree{
     int[] st ,arr;

     public Segment_tree(int[] arr,int n) {
         this.arr = Arrays.copyOf(arr,n);
         this.st= new int[4*n];
         buildTree(0,0,n-1);

     }
     public void buildTree(int i,int l,int r){
         if(l==r){
             st[i]=l;
             return;
         }
         int mid =  l + (r-l)/2;
         buildTree(2*i+1,l,mid);
         buildTree(2*i+2,mid+1,r);
         if(arr[st[2*i+1]]>=arr[st[2*i+2]])
             st[i]=st[2*i+1];
         else
             st[i]=st[2*i+2];

     }
     public int rangeMax(int i,int l, int r,int start,int end){

         if(l>end || r<start || l>r){
             return -1;
         }
         if(l>=start && r<=end)
         {
             return st[i];

         }
         int mid = l + (r-l)/2 ;
         int leftIndex=rangeMax(2*i+1,l,mid,start,end);
         int rightIndex=rangeMax(2*i+2,mid+1,r,start,end);
         if(leftIndex==-1) return rightIndex;
         if(rightIndex==-1) return leftIndex;
         if(arr[leftIndex]>=arr[rightIndex]) return leftIndex;
         return rightIndex;
     }
 }
class Solution {
  Segment_tree s;
    public int binarySearch(int i,int j,int[] arr){
        int ans=-1,n=arr.length;
        int low=i+1,high=n-1;
        while(low<=high){
            int mid= low + (high-low)/2 ;
            int res=s.rangeMax(0,0,n-1,low,mid);
            boolean valid = arr[res]>arr[i] && arr[res]>arr[j];
            if(valid){
               ans=mid;
                high=mid-1;
            }
            else{
                low=mid+1;
            }
        }
        return ans   ;
    }
    public int[] leftmostBuildingQueries(int[] heights, int[][] queries) {
        int n= heights.length,q= queries.length;
        int[] res = new int[q];int k=0;
        s = new Segment_tree(heights,n);
        for(int[] que : queries){
            int st=que[0];
            int end=que[1];
            int mx=Math.max(st,end);
            int mini=Math.min(st,end);
            if(st==end) {
                res[k++]=st;
            }
           else if(heights[mx]>heights[mini]){
            res[k++]=mx;
           }
            else{
            int idx=binarySearch(mx,mini,heights);
            res[k++]=idx;}

        }
        return res;
    }

}
