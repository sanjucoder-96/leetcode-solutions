class Solution {
       public  void generate(int row,int n,char[][] board,boolean[] checkrow,boolean[] checkcol,boolean[] upperdiag,boolean[] lowerdiag,List<List<String>> res){
        if(row==n){
            List<String> ans = new ArrayList<>();
            for(char[] r:board) ans.add(new String(r));
            res.add(ans);
            return;
        }

        for(int col=0;col<n;col++){

            if(board[row][col]=='.' && !checkcol[col] && !checkrow[row] && !upperdiag[n-row+col-1] && !lowerdiag[row+col]){
                upperdiag[n-row+col-1]=true;
                lowerdiag[row+col]=true;
                checkcol[col]=true;
                checkrow[row]=true;
                board[row][col]='Q';
                generate(row+1,n,board,checkrow,checkcol,upperdiag,lowerdiag,res);
            upperdiag[n-row+col-1]=false;
            lowerdiag[row+col]=false;
            checkcol[col]=false;
            checkrow[row]=false;
            board[row][col]='.';
            }

        }
    }
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> res = new ArrayList<>();
        char[][] board = new char[n][n];
        boolean[] checkR=new boolean[n];
        boolean[] checkC=new boolean[n];
        boolean[] lowerdiag= new boolean[2*n-1];
        boolean[] upperdiag= new boolean[2*n-1];
        Arrays.fill(checkR,false);
        Arrays.fill(checkC,false);
        Arrays.fill(lowerdiag,false);
        Arrays.fill(upperdiag,false);

        for(char[] r : board)
            Arrays.fill(r,'.');
        generate(0,n,board,checkR,checkC,upperdiag,lowerdiag,res);
        return res;
    }
}
