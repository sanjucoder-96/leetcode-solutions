## Solution

```java
class Solution {
    int[][] dp;
    public  int solve(int i,int prevIndex,int[] arr,int n){
        if(i>=n){
            return 0; 
        }
        if(dp[i][prevIndex+1]!=-1) return dp[i][prevIndex+1];
        int take=0;
        if(prevIndex==-1 || arr[prevIndex]<arr[i])
        take=1+solve(i+1,i,arr,n);
        int skip=solve(i+1,prevIndex,arr,n);
     return dp[i][prevIndex+1]= Math.max(take,skip);
    }
    public  int lengthOfLIS(int[] arr) {
        int n=arr.length;
        dp= new int[n][n+1];
        for(int[] r: dp){
            Arrays.fill(r,-1);
        }
       return solve(0,-1,arr,n);
    }

}
```
