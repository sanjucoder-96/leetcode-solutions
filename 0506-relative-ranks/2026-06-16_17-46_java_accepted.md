## Solution

```java
class Solution {
    public String[] findRelativeRanks(int[] arr) {
        int n=arr.length;
        String[] res = new String[n];
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt((int[] a)->-a[0]));
        for(int i=0;i<n;i++){
            pq.add(new int[]{arr[i],i});
        }
        int r=1;
        while(!pq.isEmpty()){
            int[] a=pq.poll();
            if(r==1){
                res[a[1]]="Gold Medal";
            }
            else if(r==2){
                res[a[1]]="Silver Medal";
            }
            else if(r==3){
                res[a[1]]="Bronze Medal";
            }
            else{
                res[a[1]]=""+r;
            }
            r++;
        }
        return res;
    }
}
```
