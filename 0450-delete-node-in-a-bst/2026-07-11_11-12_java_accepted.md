## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode findRight(TreeNode root){
        TreeNode temp=root;
        while(temp.left!=null){
            temp=temp.left;
        }
        return temp;
    }
    public TreeNode deleteNode(TreeNode root, int key) {
        if(root==null) return root;
        System.out.println(root.val+" "+(root.left)+" "+root.right+" "+key);
        if(root.val>key){
            root.left=deleteNode(root.left,key);
        }
        else if(root.val<key){
            root.right=deleteNode(root.right,key);
        }
        else {
            if(root.left==null && root.right==null) return null;
            else if(root.left==null ){
                TreeNode r =root.right;
                // root.right=null;
                return r;
            }
            else if(root.right==null) {
               return root.left;
               }
          
                TreeNode rightSmaller=findRight(root.right);
                root.val=rightSmaller.val;

                root.right=deleteNode(root.right,rightSmaller.val);
            }
        return root;
        }
    }
```
