## Solution

```java
class Solution {
    public boolean isValidSudoku(char[][] board) {
        Set<String> st = new HashSet<>();
        for(int i=0;i<9;i++){
            for(int j=0;j<9;j++){

                if(board[i][j]=='.') continue;

                String r=""+board[i][j]+"ROW"+i;
                String c=""+board[i][j]+"COL"+j;
                String box=""+board[i][j]+"BOX"+(i/3)+""+(j/3);

                if(!st.add(r) || !st.add(c) || !st.add(box)) return false;
            }
            
        }
        return true;
    }
}
```
