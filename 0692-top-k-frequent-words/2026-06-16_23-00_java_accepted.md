## Solution

```java
class Solution {
    public List<String> topKFrequent(String[] words, int k) {
         Map<String,Integer> mp = new HashMap<>();
        for(String s : words) mp.put(s,mp.getOrDefault(s,0)+1);
        List<String> res = new ArrayList<>();
         PriorityQueue<Map.Entry<String,Integer>>  pq = new PriorityQueue<>(Comparator.comparing((Map.Entry<String,Integer> e)->e.getValue()).thenComparing((a,b)->{
            return b.getKey().compareTo(a.getKey());
        })); 
        for(Map.Entry<String,Integer> s: mp.entrySet()){
            pq.add(s);
            if(pq.size()>k)
            pq.poll();
        }
        while(!pq.isEmpty()){
            res.add(pq.poll().getKey());
        }
        Collections.reverse(res);
        return res;
    }
}
```
