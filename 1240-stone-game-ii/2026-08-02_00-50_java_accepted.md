## Solution

```java
class Solution {
    int dp[][][];
    public int play(int turn ,int i,int[] piles,int M){
        int n=piles.length;
        if(i>=n) return 0;
        if(dp[turn][i][M]!=-1) return dp[turn][i][M];
        int stones=0;
        int res=(turn==1)?-1:Integer.MAX_VALUE;
        for(int x=1;x<=Math.min(2*M,n-i);x++){
            stones+=piles[i+x-1];
            if(turn==1){
                res=Math.max(res,stones+play(0,i+x,piles,Math.max(M,x)));
            }
            else
            res=Math.min(res,play(1,i+x,piles,Math.max(M,x)));
        }
        return dp[turn][i][M]=res;

    }
    public int stoneGameII(int[] piles) {
        dp= new int[2][101][101];
        for(int[][] mat : dp){
            for(int[] r : mat)
            Arrays.fill(r,-1);
        }
        return play(1,0,piles,1);
    }
}
```
