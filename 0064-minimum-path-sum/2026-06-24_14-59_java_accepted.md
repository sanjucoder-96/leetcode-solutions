## Solution

```java
class Solution {
    int[][] dp ;
   public int solve(int i,int j,int[][] arr,int m,int n){
    if(i==m-1 && j==n-1){
        return arr[i][j];
    }
    if(i>=m || j>=n) return Integer.MAX_VALUE;
    if(dp[i][j]!=-1) return dp[i][j];
    int down=solve(i+1,j,arr,m,n);
    int right=solve(i,j+1,arr,m,n);
    return dp[i][j]=arr[i][j]+Math.min(down,right);
   }
    public int minPathSum(int[][] arr) {
        int m=arr.length,n=arr[0].length;
        dp= new int[m][n];
        for(int[] r : dp)
        Arrays.fill(r,-1);
        return solve(0,0,arr,m,n);
    }
}
```
