## Solution

```java
class Solution {
    public int longestBalanced(int[] arr) {
        int n=arr.length;
        int len=0;
        int[] prefix= new int[n];
        Map<Integer,Integer> mp = new HashMap<>();
        for(int r=0;r<n;r++){
            int val=(arr[r]%2==0)?1:-1;
            int prev=mp.getOrDefault(arr[r],-1);
            if(prev!=-1){
                for(int l=0;l<=prev;l++)
                prefix[l]-=val;
            }

            for(int l=0;l<=r;l++){
                prefix[l]+=val;
            }

            for(int l=0;l<=r;l++){
                if(prefix[l]==0)
                {
                    len=Math.max(len,r-l+1);
                    break;
                }
            }

            mp.put(arr[r],r);
        } 
        return len;
    }
}
```
