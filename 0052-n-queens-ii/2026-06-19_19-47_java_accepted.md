class Solution {
     public  int  generateII(int row,int n,char[][] board,boolean[] checkrow,boolean[] checkcol,boolean[] upperdiag,boolean[] lowerdiag){
        if(row==n){
            return 1;
        }
        int count=0;
        for(int col=0;col<n;col++){

            if(board[row][col]=='0' && !checkcol[col] && !checkrow[row] && !upperdiag[n-row+col-1] && !lowerdiag[row+col]){
                upperdiag[n-row+col-1]=true;
                lowerdiag[row+col]=true;
                checkcol[col]=true;
                checkrow[row]=true;
                board[row][col]='1';
                count+=generateII(row+1,n,board,checkrow,checkcol,upperdiag,lowerdiag);
                upperdiag[n-row+col-1]=false;
                lowerdiag[row+col]=false;
                checkcol[col]=false;
                checkrow[row]=false;
                board[row][col]='0';
            }

        }
        return count;
    }
    public int totalNQueens(int n) {
        char[][] board = new char[n][n];
        boolean[] checkR=new boolean[n];
        boolean[] checkC=new boolean[n];
        boolean[] lowerdiag= new boolean[2*n-1];
        boolean[] upperdiag= new boolean[2*n-1];
        Arrays.fill(checkR,false);
        Arrays.fill(checkC,false);
        Arrays.fill(lowerdiag,false);
        Arrays.fill(upperdiag,false);

        for(char[] r : board)
            Arrays.fill(r,'0');
        return generateII(0,n,board,checkR,checkC,upperdiag,lowerdiag);
    
    }
}
