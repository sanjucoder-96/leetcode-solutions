## Solution

```java
class Solution {
    int[] parent;
    int[] rank;

    public int find(int x){
        if(parent[x]!=x) return parent[x]=find(parent[x]);
        return x;
    }
    public void union(int x,int y){
        int a=find(x),b=find(y);
        if(a==b) return;
        if(rank[a]>rank[b]){
            parent[b]=a;

        }
        else if(rank[b]>rank[a]){
            parent[a]=b;
        }
        else{
            parent[a]=b;
            rank[b]++;
        }
    }
    public boolean check(int a,int b,int c,int d){
        return a==c||b==d;
    }
    
    public int removeStones(int[][] stones) {
        int MAX=10001;
        int n=stones.length;
        int count=n;
        parent = new int[n];
        for(int i=0;i<n;i++) parent[i]=i;
        rank= new int[n];
        int[] rowMap = new int[MAX];
        int[] colMap = new int[MAX];
        Arrays.fill(rowMap,-1);
        Arrays.fill(colMap,-1);

        for(int i=0;i<n;i++){
            
            int r=stones[i][0],c=stones[i][1];
            if(rowMap[r]!=-1){
                union(i,rowMap[r]);
            }   
            else
            rowMap[r]=i;

            if(colMap[c]!=-1)
            union(i,colMap[c]);
            else 
            colMap[c]=i;
        }
        count=0;
        for(int i=0;i<n;i++){
            if(parent[i]==i) count++;
        }
        return n-count;
        
    }
}
```
