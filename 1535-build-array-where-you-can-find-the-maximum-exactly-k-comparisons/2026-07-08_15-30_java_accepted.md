## Solution

```java
class Solution {
    Integer[][][] dp;
    int MOD= (int)1e9+7;
     public int solve(int idx,int n,int m,int k,int prev){
        if(idx>=n){
            return k==0?1:0;
        }
        if(k<0) return 0;
        if(dp[idx][k][prev]!=null) return dp[idx][k][prev];
        int cnt=0;
        for(int i=1;i<=m;i++){
            if(prev==-1 || prev<i){
                cnt=(cnt+solve(idx+1,n,m,k-1,i))%MOD;
            }
            else{
                cnt=(cnt+solve(idx+1,n,m,k,prev))%MOD;
            }

        }
        return dp[idx][k][prev]=cnt%MOD;
    }
   public int numOfArrays(int n, int m, int k) {
        dp = new Integer[n][k+1][m+1];
        return solve(0,n,m,k,0);

    }
}
```
