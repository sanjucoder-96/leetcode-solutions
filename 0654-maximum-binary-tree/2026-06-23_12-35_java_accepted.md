## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode constructMaximumBinaryTree(int[] arr) {
        Deque<TreeNode> st = new ArrayDeque<>();
        
        int n=arr.length;
        for(int ele : arr){
            TreeNode curr= new TreeNode(ele);
            while(!st.isEmpty() && st.peek().val<ele){
                curr.left=st.pop();
            }
            if(!st.isEmpty()) st.peek().right=curr;

            st.push(curr);
        }
        return st.isEmpty()?null:st.removeLast();

        
        }
}
```
