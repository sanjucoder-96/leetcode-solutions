/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
 class Segment_tree{
     int[] st ,arr;

     public Segment_tree(int[] arr,int n) {
         this.arr = Arrays.copyOf(arr,n);
         this.st= new int[4*n];
        buildTree(0,0,n-1);
     }

     public void buildTree(int i,int l,int r){
        if(l==r){
            st[i]=l;
            return;
        }
        int mid= (l+r)>>1;
        buildTree(2*i+1,l,mid);
        buildTree(2*i+2,mid+1,r);
        if(arr[st[2*i+1]]>=arr[st[2*i+2]]){
            st[i]=st[2*i+1];
        }
        else st[i]=st[2*i+2];


     }

     public int rangeMax(int i,int l,int r,int start, int end){
        if(l>end || r<start){
            return -1;
        }
        if(l>=start && r<=end){
            return st[i];
        }
        int mid = (l+r)>>1;
        int leftIndex=rangeMax(2*i+1,l,mid,start,end);
        int rightIndex=rangeMax(2*i+2,mid+1,r,start,end);
        if(leftIndex==-1) return rightIndex;
        if(rightIndex==-1) return leftIndex;
      if(arr[leftIndex]>=arr[rightIndex]) return leftIndex;
         return rightIndex;
        
     }
}
class Solution {
    Segment_tree seg;
    public TreeNode construct(int l,int r,int[] arr){
       if(l>r) return null;
       int maxIdx=seg.rangeMax(0,0,arr.length-1,l,r);
       TreeNode root= new TreeNode(arr[maxIdx]);
       root.left=construct(l,maxIdx-1,arr);
       root.right=construct(maxIdx+1,r,arr);
       return root;
    }
    public TreeNode constructMaximumBinaryTree(int[] arr) {
        if(arr==null) return null;
        seg= new  Segment_tree(arr,arr.length);
        return construct(0,arr.length-1,arr);
    }
}
