class Solution {
    public boolean canShip(int[] arr,int max,int d){
        int c=1;
        int totalWeight=0;
        for(int ele : arr){
            if(totalWeight+ele<=max) totalWeight+=ele;
            else{
                totalWeight=ele;
                c++;
            }
        }
        return c<=d;
    }
    public int shipWithinDays(int[] arr, int days) {
        int n=arr.length;
        int low=-1,high=0;
        for(int ele : arr){
            low=Math.max(low,ele);
            high+=ele;
        }
        while(low<=high){
            int mid= low + (high-low)/2;
            if(canShip(arr,mid,days)){
                high=mid-1;
            }
            else low=mid+1;
            
            
                    }
                    return low;
    }
}
