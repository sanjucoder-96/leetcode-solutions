## Solution

```java
class Solution {
    public char processStr(String s, long k) {
        int n = s.length();
        long L = 0;
        for (char c : s.toCharArray()) {
            if (c == '*') {
                if (L > 0)
                    L--;
            } else if (c == '#') {
                L *= 2;
            } else if (c == '%')
                continue;
            else {
                L++;

            }
        }
        if(k>=L) return '.';
        for (int i = n - 1; i >= 0; i--) {
            char c = s.charAt(i);
            if (c == '*') {
                L++;

            } else if (c == '#') {
                L /= 2;
                k = (k >= L) ? k - L : k;

            }
            else if(c=='%'){
                k=L-k-1;
            }
            else{
                L--;
            
        }
        if(L==k) return c;
        }
        return '.';
    }
}
```
