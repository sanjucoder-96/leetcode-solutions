class Solution {
    class Pair {
        int first;
        int second;

        public Pair(int first, int second) {
            this.first = first;
            this.second = second;
        }
    }

    static int[] di = { 0, 0, -1, 1 };
    static int[] dj = { -1, 1, 0, 0 };

    public boolean check(List<List<Integer>> grid, int sf) {
        int n = grid.size();
        boolean[][] visited = new boolean[n][n];
        Queue<Pair> q = new LinkedList<>();
        q.add(new Pair(0, 0));
        visited[0][0] = true;
        if(grid.get(0).get(0) < sf)
    return false;
        while (!q.isEmpty()) {
            Pair p = q.poll();
            int i = p.first;
            int j = p.second;
            if (i == n - 1 && j == n - 1) {
                return true;
            }
            for (int k = 0; k < 4; k++) {
                int x = i + di[k];
                int y = j + dj[k];

                if (x < 0 || y < 0 || x >= n || y >= n || visited[x][y]) {
                    continue;
                }
                if (grid.get(x).get(y) < sf) {
                    continue;
                }
                visited[x][y] = true;
                q.add(new Pair(x, y));
            }
        }
        return false;

    }

    public int maximumSafenessFactor(List<List<Integer>> grid) {
        int n = grid.size();
        List<List<Integer>> safeValues = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            List<Integer> row = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                row.add(0);
            }
            safeValues.add(row);
        }
        List<List<Boolean>> visited = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            List<Boolean> row = new ArrayList<>();
            for (int j = 0; j < n; j++) {
                row.add(false);
            }
            visited.add(row);
        }
        Deque<Pair> q = new ArrayDeque<>();
        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid.get(i).size(); j++) {
                if (grid.get(i).get(j) == 1) {
                    visited.get(i).set(j, true);
                    safeValues.get(i).set(j, 0);
                    q.add(new Pair(i, j));
                }
            }
        }
        int level = 0;
        while (!q.isEmpty()) {
            int size = q.size();
            while (size-- > 0) {
                Pair p = q.poll();
                int i = p.first;
                int j = p.second;
                    safeValues.get(i).set(j, level);
                for (int k = 0; k < 4; k++) {
                    int new_i = i + di[k];
                    int new_j = j + dj[k];

                    if (new_i < 0 || new_j < 0 || new_i >= n || new_j >= n || visited.get(new_i).get(new_j))
                        continue;
                    visited.get(new_i).set(new_j, true);
                    q.add(new Pair(new_i, new_j));

                }

            }
            level++;

        }
        int low = 0, high = 400, ans = 0;
        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (check(safeValues, mid)) {
                ans = mid;
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        return ans;

    }
}
