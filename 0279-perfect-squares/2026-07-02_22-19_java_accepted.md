## Solution

```java
class Solution {
    public boolean isPerfect(int n ){
         int sq = (int)(Math.sqrt(n));
        
        return sq*sq==n;
    }
    public int minSquares(int n,int[] dp){
        if(isPerfect(n))
         return 1;
        if(dp[n]!=-1) return dp[n];
        int res=Integer.MAX_VALUE;
        for(int i=1;i*i<=n;i++){
            res=Math.min(res,minSquares(i*i,dp)+minSquares(n-(i*i),dp));
        }
        return dp[n]=res;
    }
    public int numSquares(int n) {
       int[] dp = new int[n+1];
      for(int i=1;i<=n;i++){
        if(isPerfect(i)){
            dp[i]=1;
        }
        else{
            int min=i;
            for(int j=1;j*j<=i;j++){
                min=Math.min(dp[j*j]+dp[i-j*j],min);
            }
            dp[i]=min;
        }
      }
      return dp[n];
    }
}
```
