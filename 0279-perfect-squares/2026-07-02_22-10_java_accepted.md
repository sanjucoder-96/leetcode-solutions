## Solution

```java
class Solution {
    public int minSquares(int n,int[] dp){
          int sq = (int)(Math.sqrt(n));
        
        if(sq*sq==n) return 1;
        if(dp[n]!=-1) return dp[n];
        int res=Integer.MAX_VALUE;
        for(int i=1;i*i<=n;i++){
            res=Math.min(res,minSquares(i*i,dp)+minSquares(n-(i*i),dp));
        }
        return dp[n]=res;
    }
    public int numSquares(int n) {
       int[] dp = new int[n+1];
       Arrays.fill(dp,-1);
        return minSquares(n,dp);
    }
}
```
