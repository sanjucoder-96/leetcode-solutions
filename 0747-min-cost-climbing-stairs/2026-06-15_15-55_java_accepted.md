## Solution

```java
class Solution {
    public int findCost(int i,int[] cost,int n,int[] dp){
        if(i>=n) return 0;
        if(dp[i]!=-1) return dp[i];
        int oneStep=findCost(i+1,cost,n,dp);
        int twoStep=findCost(i+2,cost,n,dp);
        return dp[i]=cost[i]+Math.min(oneStep,twoStep);
    }
    public int minCostClimbingStairs(int[] cost) {
        int n=cost.length;
        int[] dp = new int[n];
        
        for(int i=0;i<n;i++){
            int a=(i>0)?dp[i-1]:0;
            int b=(i>1)?dp[i-2]:0;
            dp[i]=cost[i]+Math.min(a,b);
        }
        return Math.min(dp[n-1],dp[n-2]);
    }
}
```
