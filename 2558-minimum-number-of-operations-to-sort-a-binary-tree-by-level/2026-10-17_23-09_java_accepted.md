## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int minSwaps(int[] arr){
        int swaps=0;
        int n=arr.length;
        Map<Integer,Integer> mp  = new HashMap<>();
        int[] copy = arr.clone();
        Arrays.sort(copy);
        for(int i=0;i<n;i++) mp.put(copy[i],i);
        boolean[] visited = new boolean[n];
        for(int i=0;i<n;i++){
            if(visited[i] ) continue;
            int j=i,len=0;
            while(!visited[j]){
                visited[j]=true;
                j=mp.get(arr[j]);
                len++;
            }
            swaps+=len-1;
        }
        return swaps;
    }
    public int minimumOperations(TreeNode root) {
            Deque<TreeNode>  dq = new ArrayDeque<>();
        dq.add(root);
        List<List<Integer>> res= new ArrayList<>();
        while(!dq.isEmpty()){
            int size=dq.size();
        List<Integer> l  = new ArrayList<>();
            while(size-->0){
                TreeNode front=dq.poll();
                l.add(front.val );
                if(front.left!=null ) dq.add(front.left);
                if(front.right!=null) dq.add(front.right);
            }
            res.add(l);
        }
        int ans=0;
        for(List<Integer> l : res){
            ans+=minSwaps(l.stream().mapToInt(Integer::intValue).toArray());
        }

        return ans;
    }
}
```
