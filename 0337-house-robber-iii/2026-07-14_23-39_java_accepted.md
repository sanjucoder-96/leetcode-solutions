## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public static int maxSum(TreeNode root,Map<TreeNode,Integer> map){
        if(root==null) return 0;
        if(map.get(root)!=null) return map.get(root);
        int take=0;
        take=root.val;
        int leftv=0;
        int rightv=0;
        if(root.left!=null){
            leftv=maxSum(root.left.left,map);
            rightv=maxSum(root.left.right,map);
        }
        int leftu=0,righu=0;
        if(root.right!=null){
            leftu=maxSum(root.right.left,map);
            righu=maxSum(root.right.right,map);
        }
        take+=leftu+leftv+rightv+righu;
        int skip=0;
        int skipleft=maxSum(root.left,map);
        int skipright=maxSum(root.right,map);

        skip=skipleft+skipright;
        int ans=Math.max(skip,take);
        map.put(root,ans);
        return ans;
    }
    public int rob(TreeNode root) {
        Map<TreeNode,Integer> mp = new HashMap<>();
        return maxSum(root,mp);
    }
}
```
