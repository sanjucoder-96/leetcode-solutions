class Solution {
   static Integer[][][][] dp;
    public static int solve(int idx,int tight,String num,int n,int mask,int lz){
        if (idx==n) return (lz==1)?0:1;
        if(dp[idx][tight][mask][lz]!=null) return dp[idx][tight][mask][lz];
        int res=0;
        int ub=(tight==1)?num.charAt(idx)-'0':9;
        int lb=0;
        for(int dig=lb;dig<=ub;dig++){
            if(lz==1  ){
            res+=solve(idx+1,(tight==1 && dig==ub)?1:0,num,n,(dig!=0)?mask| (1<<dig):mask,(lz==1 && dig==0)?1:0);
            }
            else{
                boolean used = ((mask >> dig) & 1) == 1;
                if(used) continue;
                res+=solve(idx+1,(tight==1 && dig==ub)?1:0,num,n,mask | (1<<dig),lz);
            }

        }
        return dp[idx][tight][mask][lz]=res;

    }
    public static int countSpecialNumbers(int n) {
    String s =String.valueOf(n);
    dp =   new  Integer[10][2][1024][2];
    return solve(0,1,s,s.length(),0,1);
    }
}
