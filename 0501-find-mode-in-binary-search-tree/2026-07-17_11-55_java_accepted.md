## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void inorder(TreeNode root,Map<Integer,Integer> mp,int[] max){
        if(root==null) return;
        inorder(root.left,mp,max);

        mp.put(root.val,mp.getOrDefault(root.val,0)+1);
        max[0]=Math.max(max[0],mp.get(root.val));
        inorder(root.right,mp,max);
    }
    public int[] findMode(TreeNode root) {
        Map<Integer,Integer> mp = new HashMap<>();
        int[] max={-1};
        inorder(root,mp,max);
        List<Integer> res = new ArrayList<>();
        for(int key : mp.keySet())
        {
            if(mp.get(key)==max[0]){
                res.add(key);
            }
        }
        int[] ans = new int[res.size()];
        for(int i=0;i<res.size();i++){
            ans[i]=res.get(i);
        }
        return ans;
    }
}
```
