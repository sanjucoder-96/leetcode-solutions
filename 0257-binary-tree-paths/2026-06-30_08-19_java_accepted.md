## Solution

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public void trace(TreeNode root,String s,List<String> res){
        if(root==null) return;
        
        if(root.left==null && root.right==null){
            s+=root.val;
            res.add(new String(s));
            return;
        }
        trace(root.left,s+root.val+"->",res);
        trace(root.right,s+root.val+"->",res);

    }
    public List<String> binaryTreePaths(TreeNode root) {
        List<String> res = new ArrayList<>();
        trace(root,"",res);
        return res;
    }
}
```
