class Solution {
    public String smallestSubsequence(String s) {
        int n=s.length();
        char[] arr=s.toCharArray();
        int[] lastIndex= new int[26];
        boolean[] seen = new boolean[26];
        for(int i=0;i<n;i++){
            int index=arr[i]-'a';
            lastIndex[index]=i;
        }
        Deque<Character> dq = new ArrayDeque<>();
              System.out.println(dq);
        for(int i=0;i<n;i++){
            char c=arr[i];
            int idx=c-'a';
            if(seen[idx]) continue;
            while(!dq.isEmpty() && dq.peek()>c && lastIndex[dq.peek()-'a']>i){
                seen[dq.pop()-'a']=false;
            }
            dq.push(c);
            seen[idx]=true;
        }
        StringBuilder sb = new StringBuilder();
        while(!dq.isEmpty()){
            sb.append(dq.pop());
        }
        sb.reverse();
        return sb.toString();
    }
}
